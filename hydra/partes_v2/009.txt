e", {Size = UDim2.new(1, 0, 0, 0), AutomaticSize = Enum.AutomaticSize.Y, BackgroundColor3 = C.Glass, BackgroundTransparency = nested and 0.94 or CARD_ALPHA,
        BorderSizePixel = 0, ClipsDescendants = true, LayoutOrder = order(), Parent = parent})

    mk("TextLabel", {Size = UDim2.new(1, -190, 0, 16), Position = UDim2.new(0, 14, 0.5, o.Desc and -15 or -8), BackgroundTransparency = 1, Text = o.Title or "", TextColor3 = C.Text,
        Font = Enum.Font.GothamBold, TextSize = nested and 12 or 13, TextXAlignment = Enum.TextXAlignment.Left, TextTruncate = Enum.TextTruncate.AtEnd, Parent = head})

            Font = Enum.Font.GothamSemibold, TextSize = 11, TextXAlignment = Enum.TextXAlignment.Left, TextTruncate = Enum.TextTruncate.AtEnd, Parent = head})
    end
    local pill = mk("TextButton", {Size = UDim2.fromOffset(160, 30), AnchorPoint = Vector2.new(1, 0.5), Position = UDim2.new(1, -12, 0.5, 0), BackgroundColor3 = C.Bg, BackgroundTransparency = 0.3,

    mk("UIPadding", {PaddingLeft = UDim.new(0, 10), PaddingRight = UDim.new(0, 10), PaddingBottom = UDim.new(0, 10), Parent = listWrap})
    local list = mk("ScrollingFrame", {Size = UDim2.new(1, 0, 0, 0), BackgroundTransparency = 1, BorderSizePixel = 0, ScrollBarThickness = 3, ScrollBarImageColor3 = C.Accent,
        CanvasSize = UDim2.new(), AutomaticCanvasSize = Enum.AutomaticSize.Y, Parent = listWrap})
    mk("UIListLayout", {Padding = UDim.new(0, 4), SortOrder = Enum.SortOrder.LayoutOrder, Parent = list})
    local function options()
        local raw = type(o.Options) == "function" and o.Options() or o.Options or {}
        return type(raw) == "table" and raw or {}
    end
    local function textFor(value)
        for _, option in ipairs(options()) do
            if option.value == value then return option.text or tostring(value) end
        end
        return tostring(value)
    end
    local chosen = {}
    local function has(value)
        for _, v in ipairs(chosen) do
            if v == value then return true end
        end
        return false
    end
    local function summary()
        if #chosen == 0 then return o.Placeholder or "\226\128\148" end
        if #chosen <= 2 then
            local names = {}
            for _, v in ipairs(chosen) do names[#names + 1] = textFor(v) end
            return table.concat(names, ", ")
        end
        return #chosen .. " " .. tr("seleccionados")
    end
    local function copy()
        local out = {}
        for i, v in ipairs(chosen) do out[i] = v end
        return out
    end
    local function commit(silent)
        pillText.Text = summary()
        Config.flags[id] = copy()
        saveConfig()
        if o.Callback and not silent then pcall(o.Callback, copy()) end
    end
    function obj:Set(values, silent)
        chosen = {}
        if type(values) == "table" then
            for _, v in ipairs(values) do
                if not has(v) then chosen[#chosen + 1] = v end
            end
        end
        commit(silent)
    end
    function obj:Get() return copy() end
    function obj:Refresh() pillText.Text = summary() end
    local function close()
        listWrap.Visible = false
        tween(chevron, 0.25, {Rotation = 90, TextColor3 = C.Dim})
        tween(pillStroke, 0.2, {Color = C.Stroke, Transparency = 0.5})
        tween(cardStroke, 0.2, {Color = C.Stroke, Transparency = nested and 0.65 or 0.5})
    end
    local function open()
        for _, child in ipairs(list:GetChildren()) do
            if child:IsA("GuiObject") then child:Destroy() end
        end
        local all = options()
        if #all == 0 then
            mk("TextLabel", {Size = UDim2.new(1, 0, 0, 28), BackgroundTransparency = 1, Text = "No hay nada para elegir", TextColor3 = C.Dim, Font = Enum.Font.GothamSemibold, TextSize = 11, Parent = list})
        end
        for index, option in ipairs(all) do
            local item = mk("TextButton", {Size = UDim2.new(1, -6, 0, 32), BackgroundColor3 = C.Glass, BackgroundTransparency = 0.94, AutoButtonColor = false, Text = "", LayoutOrder = index, Parent = list})
            corner(item, 9)
            local itemStroke = stroke(item, C.Accent, 1)
            local label = mk("TextLabel", {Size = UDim2.new(1, -44, 1, 0), Position = UDim2.fromOffset(12, 0), BackgroundTransparency = 1, Text = option.text or tostring(option.value),
                TextColor3 = C.Text, Font = Enum.Font.GothamBold, TextSize = 11, TextXAlignment = Enum.TextXAlignment.Left, TextTruncate = Enum.TextTruncate.AtEnd, Parent = item})
            local mark = mk("TextLabel", {Size = UDim2.fromOffset(24, 32), AnchorPoint = Vector2.new(1, 0), Position = UDim2.new(1, -8, 0, 0), BackgroundTransparency = 1, Text = "\226\156\147",
                TextColor3 = C.Accent, TextTransparency = 1, Font = Enum.Font.GothamBlack, TextSize = 13, Parent = item})
            local function paintItem()
                local on = has(option.value)
                item.BackgroundColor3 = on and C.Accent or C.Glass
                tween(item, 0.12, {BackgroundTransparency = on and 0.82 or 0.94})
                itemStroke.Transparency = on and 0.4 or 1
                label.TextColor3 = on and C.Accent or C.Text
                mark.TextTransparency = on and 0 or 1
            end
            paintItem()
            item.MouseButton1Click:Connect(function()
                if has(option.value) then
                    for i, v in ipairs(chosen) do
                        if v == option.value then table.remove(chosen, i) break end
                    end
                else
                    chosen[#chosen + 1] = option.value
                end
                paintItem()
                commit(false)
            end)
        end
        list.Size = UDim2.new(1, 0, 0, math.max(math.min(#all, 6) * 36, 28))
        listWrap.Visible = true
        tween(chevron, 0.25, {Rotation = 270, TextColor3 = C.Accent})
        tween(pillStroke, 0.2, {Color = C.Accent, Transparency = 0.2})
        tween(cardStroke, 0.2, {Color = C.Accent, Transparency = 0.45})
    end
    pill.MouseButton1Click:Connect(function()
        if listWrap.Visible then close() else open() end
    end)
    local saved = Config.flags[id]
    if type(saved) == "table" then obj:Set(saved, true) elseif type(o.Default) == "table" then obj:Set(o.Default, true) else pillText.Text = summary() end
    if #chosen > 0 and o.Callback then task.defer(function() pcall(o.Callback, copy()) end) end
    Hydra.Options[id] = obj
    return obj
end

    end
    local function parseHex(text)
        text = string.gsub(tostring(text), "[%s#]", "")
        if #text == 3 then text = string.rep(string.sub(text, 1, 1), 2) .. string.rep(string.sub(text, 2, 2), 2) .. string.rep(string.sub(text, 3, 3), 2) end
        if not string.match(text, "^%x%x%x%x%x%x$") then return nil end
