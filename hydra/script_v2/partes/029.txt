        patch("useEvent.GetTiers (candado visual)", EventSvc, "GetTiers", function(orig)
            return function(...)
                local real = orig(...)
                if type(real) ~= "table" then
                    return real
                end
                local out = {}
                for i, t in ipairs(real) do
                    local c = t
                    if type(t) == "table" and t.State == "Locked" then
                        if SHOP_MODE == "B" then
                            c = table.clone(t)
                            c.State = "Unlocked"
                            c.IsTimeOpen = true
                            c.IsWaitingOnTimer = false
                            c.ItemsMet = true
                            c.UnlockProgress = nil
                        elseif t.IsWaitingOnTimer then
                            c = table.clone(t)
                            c.IsTimeOpen = true
                            c.IsWaitingOnTimer = false
                            if t.ItemsMet then
                                c.State = "Unlocked"
                            end
                        end
                    end
                    out[i] = c
                end
                return out
            end
        end)
    end

    function M.Disable()
        gen = gen + 1
        if not active and #patches == 0 then
            return
        end
        for i = #patches, 1, -1 do
            local p = patches[i]
            pcall(function()
                p.tbl[p.key] = p.orig
            end)
        end
        patches = {}
        table.clear(fakeE)
        table.clear(fakeC)
        table.clear(fakeS)
        table.clear(lastBuild)
        notifyUI()
        modernSetOriginal = nil
        table.clear(modernInput)
        pending = nil
        emoteOn, cosOn = false, false
        UA2.skinOn, UA2.slotsOn = false, false
        pcall(function() Inventory.Changed:Fire() end)
        pcall(function() Loadout.Changed:Fire() end)
        active = false
        refreshWheel()
        log("Descargado: todo restaurado")
    end
