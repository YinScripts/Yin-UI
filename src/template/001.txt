-- ==================== OPTIMIZACIÓN (Ajustes > Optimización) ====================
-- Banderas que leen los loops de la librería para no gastar CPU/GPU en equipos que no la aguantan.
-- Se guardan en un archivo aparte para no tocar el formato del config principal.
local PerfOpt = {
    File = "YinYang_Optimizacion.json",
    Flags = {
        NoAnimations = false,      -- tweens de la interfaz instantáneos y loops infinitos pausados
        NoTextEffects = false,     -- sin el loop de efectos de texto (se guarda el elegido)
        StaticBackground = false,  -- fondo animado (sprite/video) quieto
        NoLogoAnim = false,        -- logo flotante sin animación por frame
        NoHomeAvatar = false,      -- avatar/owner de Inicio quietos
        NoDecorAnim = false,       -- título, banner, ecualizador y vistas previas sin animar
        NoUISounds = false,        -- sin sonidos de clic
        LiteShaders = false,       -- shaders en calidad Lite
        NoPostFX = false,          -- sin profundidad de campo, blur ni rayos de sol
        NoShaders = false,         -- shaders apagados (también al iniciar)
        NoShadows = false,         -- sin sombras globales del juego
        NoParticles = false,       -- sin partículas, estelas, humo ni fuego del mapa
        MapBoost = false,          -- mapa liviano: plástico liso, sin texturas ni decoración
        LowGraphics = false,       -- calidad de render al mínimo (si el executor lo permite)
        FpsCap30 = false,          -- límite de 30 FPS (si el executor tiene setfpscap)
    },
    PrevQuality = nil,
    Hooks = {},
    Loops = {},
    World = { PostFX = {}, Particles = {}, Parts = {}, Decals = {}, Appearances = {}, Token = 0 },
    InstantInfo = TweenInfo.new(0),
    ParticleClasses = { ParticleEmitter = true, Trail = true, Beam = true, Smoke = true, Fire = true, Sparkles = true },
    PostFXClasses = { DepthOfFieldEffect = true, BlurEffect = true, SunRaysEffect = true },
}

function PerfOpt.IsUI(instance)
    if typeof(instance) ~= "Instance" then return false end
    local ok, result = pcall(function()
        return instance:IsA("GuiObject") or instance:IsA("UIComponent")
    end)
    return ok and result == true
end

function PerfOpt.Save()
    if type(writefile) ~= "function" then return end
    pcall(function()
        local data = {}
        for key, value in pairs(PerfOpt.Flags) do data[key] = value end
        data._prevQuality = PerfOpt.PrevQuality
        writefile(PerfOpt.File, HttpService:JSONEncode(data))
    end)
end

function PerfOpt.Load()
    if type(readfile) ~= "function" or type(isfile) ~= "function" then return end
    pcall(function()
        if not isfile(PerfOpt.File) then return end
        local data = HttpService:JSONDecode(readfile(PerfOpt.File))
        if type(data) ~= "table" then return end
        for key in pairs(PerfOpt.Flags) do
            if type(data[key]) == "boolean" then PerfOpt.Flags[key] = data[key] end
        end
        if type(data._prevQuality) == "string" then PerfOpt.PrevQuality = data._prevQuality end
    end)
end
PerfOpt.Load()

-- Cambia una bandera, ejecuta su gancho (si lo hay) y guarda
function PerfOpt.Set(key, on)
    if PerfOpt.Flags[key] == nil then return end
    PerfOpt.Flags[key] = on == true
    local hook = PerfOpt.Hooks[key]
    if hook then pcall(hook, on == true) end
    PerfOpt.Save()
end

-- Loops infinitos de la interfaz (brillos, giros, pulsos): se pausan con "Sin animaciones" y vuelven al apagarlo
function PerfOpt.SetLoopsPaused(paused)
    for index = #PerfOpt.Loops, 1, -1 do
        local entry = PerfOpt.Loops[index]
        local alive = entry.instance and entry.instance.Parent ~= nil
        if not alive then
            table.remove(PerfOpt.Loops, index)
        elseif not entry.wanted then
            continue
        elseif paused then
            pcall(function() entry.real:Pause() end)
        else
            pcall(function() entry.real:Play() end)
        end
    end
end

function PerfOpt.WrapLoop(real, instance)
    local entry = { real = real, instance = instance, wanted = false }
    table.insert(PerfOpt.Loops, entry)
    local methods = {
        Play = function()
            entry.wanted = true
            if not PerfOpt.Flags.NoAnimations then real:Play() end
        end,
        Pause = function() entry.wanted = false; real:Pause() end,
        Cancel = function() entry.wanted = false; real:Cancel() end,
        Destroy = function() entry.wanted = false; real:Destroy() end,
    }
    return setmetatable({}, {
        __index = function(_, key)
            local method = methods[key]
            if method then return function() return method() end end
            local value = real[key]
            if type(value) == "function" then
                return function(_, ...) return value(real, ...) end
            end
            return value
        end,
    })
end

-- Envoltorio de TweenService: misma API (:Create), pero respeta "Sin animaciones" en la interfaz
local TweenService = {}
function TweenService:Create(instance, info, goals)
    if PerfOpt.IsUI(instance) then
        if typeof(info) == "TweenInfo" and info.RepeatCount < 0 then
            return PerfOpt.WrapLoop(RealTweenService:Create(instance, info, goals), instance)
        end
        if PerfOpt.Flags.NoAnimations then
            info = PerfOpt.InstantInfo
        end
    end
    return RealTweenService:Create(instance, info, goals)
end

function PerfOpt.IsOwnInstance(instance)
    local node = instance
    while node and node ~= workspace and node ~= game do
        if string.sub(node.Name or "", 1, 14) == "YinYangShader_" then return true end
        node = node.Parent
    end
    return false
end

-- ----- Mundo: sombras -----
function PerfOpt.ApplyShadows()
    local W = PerfOpt.World
    local Lighting = game:GetService("Lighting")
    if PerfOpt.Flags.NoShadows then
        pcall(function()
            -- si alguien (un preset) las prendió, ese es el valor a devolver al apagar la opción
            if W.Shadows == nil or Lighting.GlobalShadows == true then W.Shadows = Lighting.GlobalShadows end
            Lighting.GlobalShadows = false
        end)
    elseif W.Shadows ~= nil then
        pcall(function() Lighting.GlobalShadows = W.Shadows end)
        W.Shadows = nil
    end
end

-- ----- Mundo: post-procesado pesado (profundidad de campo, blur, rayos) -----
function PerfOpt.ApplyPostFX()
    local W = PerfOpt.World
    local containers = { game:GetService("Lighting"), workspace.CurrentCamera }
    if PerfOpt.Flags.NoPostFX then
        for _, container in ipairs(containers) do
            if container then
                for _, fx in ipairs(container:GetChildren()) do
                    if PerfOpt.PostFXClasses[fx.ClassName] then
                        pcall(function()
                            if W.PostFX[fx] == nil or fx.Enabled == true then W.PostFX[fx] = fx.Enabled end
                            fx.Enabled = false
                        end)
                    end
                end
            end
        end
    else
        for fx, was in pairs(W.PostFX) do
            if fx.Parent then pcall(function() fx.Enabled = was end) end
        end
        W.PostFX = {}
    end
end

-- ----- Mundo: partículas, estelas, humo y fuego -----
function PerfOpt.DisableParticle(instance)
    if not PerfOpt.Flags.NoParticles or not instance or not PerfOpt.ParticleClasses[instance.ClassName] then return end
    if PerfOpt.IsOwnInstance(instance) then return end
    local W = PerfOpt.World
    pcall(function()
        if W.Particles[instance] == nil then W.Particles[instance] = instance.Enabled end
        instance.Enabled = false
    end)
end

function PerfOpt.ApplyParticles()
    local W = PerfOpt.World
    if W.ParticleConn then W.ParticleConn:Disconnect(); W.ParticleConn = nil end
    if PerfOpt.Flags.NoParticles then
        task.spawn(function()
            for index, instance in ipairs(workspace:GetDescendants()) do
                if not PerfOpt.Flags.NoParticles then return end
                PerfOpt.DisableParticle(instance)
                if index % 400 == 0 then task.wait() end
            end
        end)
        W.ParticleConn = workspace.DescendantAdded:Connect(function(instance)
            if PerfOpt.ParticleClasses[instance.ClassName] then task.defer(PerfOpt.DisableParticle, instance) end
        end)
    else
        for instance, was in pairs(W.Particles) do
            if instance.Parent then pcall(function() instance.Enabled = was end) end
        end
        W.Particles = {}
    end
end

-- ----- Mundo: mapa liviano (reversible). No toca avatares ni los efectos propios de shaders -----
function PerfOpt.LightenPart(part)
    if not PerfOpt.Flags.MapBoost or not part or not part.Parent or not part:IsA("BasePart") then return end
    if PerfOpt.IsOwnInstance(part) then return end
    local model = part:FindFirstAncestorOfClass("Model")
    if model and model:FindFirstChildOfClass("Humanoid") then return end
    local W = PerfOpt.World
    pcall(function()
        local material = part.Material
        if material ~= Enum.Material.Neon and material ~= Enum.Material.Glass and material ~= Enum.Material.ForceField then
            if W.Parts[part] == nil then W.Parts[part] = { Material = material, CastShadow = part.CastShadow } end
            part.Material = Enum.Material.SmoothPlastic
            part.CastShadow = false
        end
        pcall(function()
            if part.MaterialVariant ~= "" then
                W.Parts[part] = W.Parts[part] or { CastShadow = part.CastShadow }
                W.Parts[part].MaterialVariant = part.MaterialVariant
                part.MaterialVariant = ""
            end
        end)
    end)
    for _, child in ipairs(part:GetChildren()) do
        if child:IsA("Texture") or child:IsA("Decal") then
            pcall(function()
                if child.Transparency < 1 then
                    W.Decals[child] = child.Transparency
                    child.Transparency = 1
                end
            end)
        elseif child:IsA("SurfaceAppearance") then
            W.Appearances[child] = part
            pcall(function() child.Parent = nil end)
        end
    end
end

function PerfOpt.ApplyMapBoost()
    local W = PerfOpt.World
    W.Token = W.Token + 1
    local token = W.Token
    if W.MapConn then W.MapConn:Disconnect(); W.MapConn = nil end
    local terrain = workspace:FindFirstChildOfClass("Terrain")
    if PerfOpt.Flags.MapBoost then
        if terrain then
            pcall(function()
                if W.Terrain == nil then
                    W.Terrain = { Decoration = terrain.Decoration, WaterWaveSize = terrain.WaterWaveSize, WaterWaveSpeed = terrain.WaterWaveSpeed, WaterReflectance = terrain.WaterReflectance }
                end
                terrain.Decoration = false
                terrain.WaterWaveSize = 0
                terrain.WaterWaveSpeed = 0
                terrain.WaterReflectance = 0
            end)
        end
        task.spawn(function()
            for index, instance in ipairs(workspace:GetDescendants()) do
                if token ~= W.Token then return end
                if instance:IsA("BasePart") then PerfOpt.LightenPart(instance) end
                if index % 300 == 0 then task.wait() end
            end
        end)
        W.MapConn = workspace.DescendantAdded:Connect(function(instance)
            if instance:IsA("BasePart") then task.defer(PerfOpt.LightenPart, instance) end
        end)
    else
        for part, record in pairs(W.Parts) do
            if part.Parent then
                pcall(function()
                    if record.Material then part.Material = record.Material end
                    if record.MaterialVariant then part.MaterialVariant = record.MaterialVariant end
                    if record.CastShadow ~= nil then part.CastShadow = record.CastShadow end
                end)
            end
        end
        for decal, transparency in pairs(W.Decals) do
            if decal.Parent and decal.Transparency >= 1 then pcall(function() decal.Transparency = transparency end) end
        end
        for appearance, part in pairs(W.Appearances) do
            if part.Parent then pcall(function() appearance.Parent = part end) end
        end
        if terrain and W.Terrain then
            pcall(function()
                for property, value in pairs(W.Terrain) do terrain[property] = value end
            end)
        end
        W.Parts, W.Decals, W.Appearances, W.Terrain = {}, {}, {}, nil
    end
end

-- ----- Calidad de render y límite de FPS (depende del executor; todo protegido) -----
function PerfOpt.ApplyGraphics()
    local W = PerfOpt.World
    if type(settings) ~= "function" then return end
    if PerfOpt.Flags.LowGraphics then
        pcall(function()
            local rendering = settings().Rendering
            if W.Quality == nil then W.Quality = rendering.QualityLevel end
            rendering.QualityLevel = Enum.QualityLevel.Level01
        end)
    elseif W.Quality ~= nil then
        pcall(function() settings().Rendering.QualityLevel = W.Quality end)
        W.Quality = nil
    end
end

function PerfOpt.ApplyFpsCap()
    local W = PerfOpt.World
    if type(setfpscap) ~= "function" then return end
    if PerfOpt.Flags.FpsCap30 then
        if W.FpsBefore == nil then
            local ok, current = pcall(function() return type(getfpscap) == "function" and getfpscap() or nil end)
            W.FpsBefore = (ok and tonumber(current)) or 60
        end
        pcall(setfpscap, 30)
    elseif W.FpsBefore ~= nil then
        pcall(setfpscap, W.FpsBefore)
        W.FpsBefore = nil
    end
end

-- Después de aplicar un shader: vuelve a apagar lo que el usuario pidió apagar
function PerfOpt.AfterShaders()
    if PerfOpt.Flags.NoShadows then PerfOpt.ApplyShadows() end
    if PerfOpt.Flags.NoPostFX then PerfOpt.ApplyPostFX() end
end

PerfOpt.Hooks.NoAnimations = function(on) PerfOpt.SetLoopsPaused(on) end
PerfOpt.Hooks.NoShadows = function() PerfOpt.ApplyShadows() end
PerfOpt.Hooks.NoPostFX = function() PerfOpt.ApplyPostFX() end
PerfOpt.Hooks.NoParticles = function() PerfOpt.ApplyParticles() end
PerfOpt.Hooks.MapBoost = function() PerfOpt.ApplyMapBoost() end
PerfOpt.Hooks.LowGraphics = function() PerfOpt.ApplyGraphics() end
PerfOpt.Hooks.FpsCap30 = function() PerfOpt.ApplyFpsCap() end

    end
end)

function YLang.Apply()
    local newLanguage = YLang.Mode
    if newLanguage == "auto" then
        newLanguage = YLang.System
    end
    if newLanguage == LibraryLanguage then
        return false
    end

    LibraryLanguage = newLanguage
    LibraryTranslationCache = {}

    if RefreshLibraryLanguage then
        task.defer(RefreshLibraryLanguage)
    end

    for _, listener in ipairs(YLang.Listeners) do
        task.spawn(pcall, listener, LibraryLanguage)
    end

    return true
end

local FloatPosFile = "Yin_Yang_FloatPos.json"
local FloatPos = { Data = {} }
pcall(function()
    if type(readfile) == "function" and type(isfile) == "function" and isfile(FloatPosFile) then
        local decoded = HttpService:JSONDecode(readfile(FloatPosFile))
        if type(decoded) == "table" then FloatPos.Data = decoded end

function FloatPos.Apply(id, win)
    local saved = FloatPos.Data[id]
    if not (win and type(saved) == "table" and type(saved.x) == "number" and type(saved.y) == "number") then return end
    local size = win.Size
    win.Position = UDim2.new(
        math.clamp(saved.x, 0, 1), -size.X.Offset / 2,
        math.clamp(saved.y, 0, 1), -size.Y.Offset / 2
    )
end

function FloatPos.Save(id, win, gui)
    if not (win and win.Parent and gui) then return end
    local viewport = gui.AbsoluteSize
    if viewport.X <= 0 or viewport.Y <= 0 then return end
    local cx = (win.AbsolutePosition.X - gui.AbsolutePosition.X + win.AbsoluteSize.X / 2) / viewport.X
    local cy = (win.AbsolutePosition.Y - gui.AbsolutePosition.Y + win.AbsoluteSize.Y / 2) / viewport.Y
    FloatPos.Data[id] = { x = math.clamp(cx, 0, 1), y = math.clamp(cy, 0, 1), locked = true }
    pcall(function()
        if type(writefile) == "function" then
            writefile(FloatPosFile, HttpService:JSONEncode(FloatPos.Data))
        end
    end)
end

function FloatPos.IsLocked(id)
    local saved = FloatPos.Data[id]
    return type(saved) == "table" and saved.locked == true
end

function FloatPos.SetLocked(id, locked)
    local saved = FloatPos.Data[id]
    if type(saved) ~= "table" then return end
    saved.locked = locked and true or false
    pcall(function()
        if type(writefile) == "function" then
            writefile(FloatPosFile, HttpService:JSONEncode(FloatPos.Data))
        end
    end)
end
