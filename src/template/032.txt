    local function fmtNum(v, step)
        if step >= 1 then return tostring(math.floor(v + 0.5)) end
        if step >= 0.1 then return string.format("%.1f", v) end
        return string.format("%.2f", v)
    end
    local function toHex(c)
        return string.format("#%02X%02X%02X", math.floor(c.R * 255 + 0.5), math.floor(c.G * 255 + 0.5), math.floor(c.B * 255 + 0.5))
    end
    local function fromHex(s)
        s = tostring(s or ""):gsub("#", ""):gsub("%s", "")
        if not s:match("^%x%x%x%x%x%x$") then return nil end
        return Color3.fromRGB(tonumber(s:sub(1, 2), 16), tonumber(s:sub(3, 4), 16), tonumber(s:sub(5, 6), 16))
    end

                ImageColor3 = def.Color or WHITE,
                ImageTransparency = def.Transparency or 0,
                ZIndex = zBase + i,
            })
            mk("UIAspectRatioConstraint", { AspectRatio = 1 }, lbl)
            out[i] = { label = lbl, def = def, state = { angle = 0, speed = def.Speed or 60, time = 0 } }
        end
        return out
    end
    local function stepLayers(entries, dt)
        for _, entry in ipairs(entries) do
            local lbl, def, st = entry.label, entry.def, entry.state
            if lbl.Parent then
                local mv = def.Movement or "None"
                local spd = def.Speed or 60
                local amt = def.Amount or 0
                local dir = def.Direction or 1
                local sc = def.Scale or 1.6
                local off = -((sc - 1) / 2)
                st.time = st.time + dt
                if mv == "Spin" then
                    if (def.Style or "Constant") == "Organic" then
                        local target = spd + math.sin(st.time * 0.8) * (spd * 0.33)
                        st.speed = st.speed + (target - st.speed) * 0.08
                    else
                        st.speed = spd
                    end
                    st.angle = st.angle + st.speed * dir * dt
                    lbl.Rotation = st.angle
                elseif mv == "Pulse" then
                    local scale = 1 + math.sin(st.time * (2 * math.pi / spd)) * ((amt ~= 0 and amt or 12) / 100)
                    lbl.Size = UDim2.new(sc * scale, 0, sc * scale, 0)
                    lbl.Position = UDim2.new(off * scale, 0, off * scale, 0)
                elseif mv == "Orbit" then
                    local radius = amt ~= 0 and amt or 6
                    local angle = st.time * math.rad(spd) * dir
                    lbl.Position = UDim2.new(off, math.cos(angle) * radius, off, math.sin(angle) * radius)
                elseif mv == "Float" then
                    lbl.Position = UDim2.new(off, 0, off, math.sin(st.time * (2 * math.pi / spd)) * (amt ~= 0 and amt or 4))
                elseif mv == "Shake" then
                    local intensity = amt ~= 0 and amt or 2
                    lbl.Position = UDim2.new(off, math.sin(st.time * spd * math.pi * 2 + 0.3) * intensity, off, math.cos(st.time * spd * math.pi * 2) * intensity)
                elseif mv == "Fade" then
                    local maxAlpha = math.clamp(amt ~= 0 and amt or 0.6, 0, 1)
                    lbl.ImageTransparency = ((math.sin(st.time * (2 * math.pi / spd)) + 1) / 2) * maxAlpha
                end
            end
        end
    end

    -- catálogo/guardado -> estado editable (colores Color3, sonidos siempre presentes)
    local function recFromIcon(icon, name)
        local r = blankRec()
        r.Name = name or ""
        r.NoBackground = icon.NoBackground == true
        r.BgColor = typeof(icon.BgColor) == "Color3" and icon.BgColor or (type(icon.BgColor) == "table" and arrayToColor(icon.BgColor, nil)) or nil
        r.BorderColor = typeof(icon.BorderColor) == "Color3" and icon.BorderColor or (type(icon.BorderColor) == "table" and arrayToColor(icon.BorderColor, nil)) or nil
        r.Layers = {}
        for i, src in ipairs(icon.Layers or {}) do
            if i > 3 then break end
            if type(src.Image) == "string" and src.Image ~= "" then
                local lay = blankLayer(src.Image)
                lay.Movement = CL.Moves[src.Movement] and src.Movement or "None"
                lay.Speed = tonumber(src.Speed) or (MOVE_INFO[lay.Movement].speed and MOVE_INFO[lay.Movement].speed[4]) or 60
                lay.Amount = tonumber(src.Amount) or 0
                lay.Direction = src.Direction == -1 and -1 or 1
                lay.Style = src.Style == "Constant" and "Constant" or "Organic"
                lay.Scale = tonumber(src.Scale) or 1.6
                lay.Transparency = tonumber(src.Transparency) or 0
                lay.Color = typeof(src.Color) == "Color3" and src.Color or (type(src.Color) == "table" and arrayToColor(src.Color, nil)) or nil
                table.insert(r.Layers, lay)
            end
        end
        if #r.Layers == 0 then r.Layers = { blankLayer("") } end
        if type(icon.ClickSound) == "table" and type(icon.ClickSound.Id) == "string" then
            r.ClickSound = { Id = icon.ClickSound.Id, Volume = tonumber(icon.ClickSound.Volume) or 0.6 }
        end
        if type(icon.IdleSound) == "table" and type(icon.IdleSound.Id) == "string" then
            r.IdleSound = { Id = icon.IdleSound.Id, Interval = tonumber(icon.IdleSound.Interval) or 15, Volume = tonumber(icon.IdleSound.Volume) or 0.15 }
        end
        return r
    end
    -- estado editable -> registro para guardar (pasa por CL.Sanitize)
    local function recToSaved(r, id)
        local s = {
            Id = id,
            Name = r.Name,
            NoBackground = r.NoBackground,
            BgColor = r.BgColor and colorToArray(r.BgColor) or nil,
            BorderColor = r.BorderColor and colorToArray(r.BorderColor) or nil,
            Created = os.time(),
            Layers = {},
            ClickSound = r.ClickSound,
            IdleSound = r.IdleSound,
        }
        for i, lay in ipairs(r.Layers) do
            local c = table.clone(lay)
            c.Color = lay.Color and colorToArray(lay.Color) or nil
            s.Layers[i] = c
        end
        return CL.Sanitize(s)
    end
    -- capas que se pueden dibujar ya (las que tienen imagen)
    local function drawableIcon(r)
        local icon = { NoBackground = r.NoBackground, BgColor = r.BgColor, BorderColor = r.BorderColor, Layers = {} }
        for _, lay in ipairs(r.Layers) do
            local img = CL.CleanAsset(lay.Image)
            if img ~= "" then
                local c = table.clone(lay)
                c.Image = img
                table.insert(icon.Layers, c)
            end
        end
        return icon
    end
