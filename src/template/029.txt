      self.Enabled, self.StormEnabled = false, false

    end

    -- Silencio total: todos los sonidos del juego (y de la librería) a volumen 0, guardando el original.
    -- Si el juego intenta subirles el volumen, se vuelven a silenciar al instante. Al salir, todo vuelve.
    function ShaderFootage:_muteSound(sound)
        if not self.Enabled or not sound or self.Muted[sound] ~= nil then return end
        if PerfOpt.IsOwnInstance(sound) then return end
        local ok = pcall(function()
            self.Muted[sound] = sound.Volume
            sound.Volume = 0
        end)
        if not ok then self.Muted[sound] = nil return end
        local okSignal, connection = pcall(function()
            return sound:GetPropertyChangedSignal("Volume"):Connect(function()
                if self.Enabled and self.Muted[sound] ~= nil and sound.Volume > 0 then
                    self.Muted[sound] = sound.Volume
                    sound.Volume = 0
                end
            end)
        end)
        if okSignal and connection then table.insert(self.MuteConnections, connection) end
    end

    function ShaderFootage:_muteAll()
        self.MuteToken = self.MuteToken + 1
        local token = self.MuteToken
        local okAdded, added = pcall(function()
            return game.DescendantAdded:Connect(function(instance)
                if instance:IsA("Sound") then task.defer(function() self:_muteSound(instance) end) end
            end)
        end)
        if okAdded and added then table.insert(self.MuteConnections, added) end
        task.spawn(function()
            local okList, list = pcall(function() return game:GetDescendants() end)
            if not okList then return end
            for index, instance in ipairs(list) do
                if token ~= self.MuteToken or not self.Enabled then return end
                local isSound = false
                pcall(function() isSound = instance:IsA("Sound") end)
                if isSound then self:_muteSound(instance) end
                if index % 500 == 0 then task.wait() end
            end
        end)
    end

    function ShaderFootage:_unmuteAll()
        self.MuteToken = self.MuteToken + 1
        for _, connection in ipairs(self.MuteConnections) do pcall(function() connection:Disconnect() end) end
        self.MuteConnections = {}
        for sound, volume in pairs(self.Muted) do
            if sound.Parent then pcall(function() sound.Volume = volume end) end
        end
        self.Muted = {}
    end

    end

    function ShaderFootage:_step(delta)
        local camera = workspace.CurrentCamera
        if not camera then return end
        -- lente abierto (se acerca suave al FOV objetivo)
        pcall(function() camera.FieldOfView = camera.FieldOfView + (self.TargetFov - camera.FieldOfView) * math.min(1, delta * 4) end)
        -- desenfoque de movimiento: cuanto más rápido gira la cámara, más borroso
        local look = camera.CFrame.LookVector
        local turn = 0
        if self.LastLook then
            turn = math.acos(math.clamp(self.LastLook:Dot(look), -1, 1)) / math.max(delta, 1 / 240)
        end
        self.LastLook = look
        local glitchBoost = self.GlitchT and 6 or 0
        local targetBlur = PerfOpt.Flags.NoPostFX and 0 or (self.BaseBlur + math.min(14, turn * 2.6) + glitchBoost)
        self.Blur = self.Blur + (targetBlur - self.Blur) * math.min(1, delta * (targetBlur > self.Blur and 18 or 6))
        local blur = ShaderManager:GetTarget("BlurEffect", false)
        if blur then pcall(function() blur.Enabled = self.Blur > 0.3; blur.Size = self.Blur end) end
        -- auto-exposición: mirar hacia los tubos del techo baja la exposición, como una cámara barata
        local base = self.BaseExposure or 0
        local exposure = base - math.max(0, look.Y) * 0.55 + math.sin(os.clock() * 0.7) * 0.03
        pcall(function()
            ShaderLightingService.ExposureCompensation = ShaderLightingService.ExposureCompensation + (exposure - ShaderLightingService.ExposureCompensation) * math.min(1, delta * 2.2)
        end)
        -- la parte "de cinta" corre a ~24 cuadros por segundo, como el video
        self.Acc = self.Acc + delta
        if self.Acc < 1 / 24 then return end
        self.Acc = 0
        if not PerfOpt.Flags.NoAnimations then
            for _, dot in ipairs(self.Grain) do
                dot.Position = UDim2.fromScale(self.Rng:NextNumber(), self.Rng:NextNumber())
                dot.BackgroundTransparency = self.Rng:NextNumber(0.62, 0.9)
            end
            if self.Lines then
                self.Lines.Position = UDim2.new(0, 0, 0, -((os.clock() * 9) % 8))
            end
        end
        if self.Flicker then self.Flicker.BackgroundTransparency = self.Rng:NextNumber(0.9, 0.985) end
        -- tirón de tracking cada 6-14 s: una banda clara baja por la pantalla y todo se borronea un instante
        local now = os.clock()
        if not self.GlitchT and now >= self.NextGlitch then
            self.GlitchT = now
            self.NextGlitch = now + self.Rng:NextNumber(6, 14)
        end
        if self.GlitchT and self.Band then
            local t = (now - self.GlitchT) / 0.45
            if t >= 1 then
                self.GlitchT = nil
                self.Band.BackgroundTransparency = 1
            else
                self.Band.Position = UDim2.new(0, 0, -0.1 + t * 1.2, 0)
                self.Band.BackgroundTransparency = 0.2 + t * 0.5
            end
        end
    end

    function ShaderFootage:SetEnabled(enabled)
        enabled = enabled == true
        if enabled == self.Enabled then return end
        self.Enabled = enabled
        local camera = workspace.CurrentCamera
        if enabled then
            if camera then self.SavedFov = camera.FieldOfView end
            -- base propia (la del preset): así la auto-exposición no se "acumula" con otros ajustes
            local preset = ShaderManager.Presets.Level0
            self.BaseExposure = preset and preset.Lighting and preset.Lighting.ExposureCompensation or 0.25
            self.LastLook, self.Blur, self.Acc, self.GlitchT = nil, self.BaseBlur, 0, nil
            self.NextGlitch = os.clock() + 4
            self.Gui = self:_buildGui()
            self:_muteAll()
            self:_startAmbient()
            self.Connection = RunService.RenderStepped:Connect(function(delta)
                local ok = pcall(self._step, self, delta)
                if not ok then return end
            end)
            return
        end
        if self.Connection then self.Connection:Disconnect(); self.Connection = nil end
        self:_unmuteAll()
        self:_stopAmbient()
        if self.Gui then pcall(function() self.Gui:Destroy() end) end
        self.Gui, self.Lines, self.Flicker, self.Band, self.Grain = nil, nil, nil, nil, {}
        if camera and self.SavedFov then pcall(function() camera.FieldOfView = self.SavedFov end) end
        self.SavedFov = nil
        -- la exposición no se toca acá: la pone el modo siguiente (o "Restaurar original")
    end

    shaderQueuePersistence = function()
        if type(writefile) ~= "function" then return end
        ShaderPersistenceToken = ShaderPersistenceToken + 1
        local token = ShaderPersistenceToken
        task.delay(0.35, function()
            if token ~= ShaderPersistenceToken then return end
            pcall(function()
                game:GetService("HttpService"):JSONEncode(shaderCapturePersistence())
            end)
            pcall(function()
                writefile(ShaderPersistenceFile, game:GetService("HttpService"):JSONEncode(shaderCapturePersistence()))
            end)
        end)
    end

    local function shaderLoadPersistence()
        if type(readfile) ~= "function" or type(isfile) ~= "function" then return nil end
        local ok, state = pcall(function()
            if not isfile(ShaderPersistenceFile) then return nil end
            return game:GetService("HttpService"):JSONDecode(readfile(ShaderPersistenceFile))
        end)
        if ok and type(state) == "table" and state.version == 1 then return state end
        return nil
    end

    local function shaderSyncImmersivePreset(name)
        ShaderReflections.OilSheen = false
        local mirrorPreset = name == "OilWet" or name == "NeonNight"
        local rainPreset = ShaderRain.SkyPresets[name] == true
        if not mirrorPreset then
            if ShaderReflections.ObjectMirrorEnabled then ShaderReflections:SetObjectMirrorEnabled(false) end
            if ShaderReflections.StripTextures then ShaderReflections:SetStripTexturesEnabled(false) end
            if ShaderReflections.SkyMirrorEnabled then ShaderReflections:SetSkyMirrorEnabled(false) end
        end
        -- la cámara de cinta solo vive en "Nivel 0"
        ShaderFootage:SetEnabled(name == "Level0")
        -- la silueta negra solo vive en depthstyle (antes quedaba prendida al pasar a otro modo)
        if name ~= "depthstyle" then ShaderSilhouette:SetEnabled(false) end
        if name == "depthstyle" then
            ShaderRain:Destroy()
            ShaderReflections:SetCharacterSpecularEnabled(false)
            ShaderReflections:SetContactEnabled(false)
            ShaderReflections:SetShadowEnabled(false)
            ShaderReflections:SetEnabled(false)
            ShaderSilhouette:SetEnabled(true)
        elseif rainPreset then
            local heavy = name ~= "Rain"
            ShaderReflections:SetCharacterSpecularEnabled(false)
            ShaderReflections:SetWetness(name == "Monsoon" and 1 or 0.92)
            ShaderReflections:SetIntensity(name == "Monsoon" and 0.42 or (heavy and 0.36 or 0.30))
            ShaderReflections:SetEnabled(true)
            ShaderReflections:SetContactEnabled(true)
            ShaderReflections:SetShadowStrength(0.24)
            ShaderReflections:SetShadowEnabled(true)
            ShaderRain.StormGapMin, ShaderRain.StormGapMax = (name == "Monsoon" and 3 or 5), (name == "Monsoon" and 7 or 12)
            ShaderRain:SetIntensity(name == "Monsoon" and 1 or (heavy and 0.95 or 0.72))
            ShaderRain:SetEnabled(true)
            ShaderRain:SetStormEnabled(heavy)
        elseif name == "GoldenSpecular" then
            ShaderRain:Destroy()
            if not PerfOpt.Flags.LiteShaders then shaderSetQuality("Cinematic") end
            ShaderReflections:ApplyGoldenFinish()
        elseif name == "OilWet" then
            ShaderRain:Destroy()
            if not PerfOpt.Flags.LiteShaders then shaderSetQuality("Cinematic") end
            ShaderReflections:ApplyOilFinish()
        elseif name == "NeonNight" then
            ShaderRain:Destroy()
            if not PerfOpt.Flags.LiteShaders then shaderSetQuality("Cinematic") end
            ShaderReflections:ApplyMirrorFinish()
        elseif name == "Photoreal" then
            ShaderRain:Destroy()
            ShaderReflections:SetCharacterSpecularEnabled(false)
            ShaderReflections:SetSurfaceMode("Floors")
            ShaderReflections:SetSceneCoverage(true)
            ShaderReflections:SetIntensity(0.12)
            ShaderReflections:SetWetness(0.5)
            ShaderReflections:SetEnabled(true)
            ShaderReflections:SetContactEnabled(true)
            ShaderReflections:SetShadowStrength(0.36)
            ShaderReflections:SetShadowEnabled(true)
        else
            ShaderRain:Destroy()
            ShaderReflections:SetCharacterSpecularEnabled(false)
            ShaderReflections:SetContactEnabled(false)
            ShaderReflections:SetEnabled(false)
            -- profundidad extra (sombras de contacto en modelos) para los modos de día y de cine
            local depthModes = {Morning = true, Midday = true, Sunset = true, Cinematic = true, Desert = true, BlueHour = true}
            if depthModes[name] then
                ShaderReflections:SetShadowStrength(0.30)
                ShaderReflections:SetShadowEnabled(true)
            end
        end
    end

    local function shaderApplyPreset(name, tween)
        local report = ShaderManager:ApplyPreset(name, {
            tween = tween == true,
            duration = 1,
        })
        if report.ok then
            shaderClearCustomOverrides()
            shaderSyncImmersivePreset(name)
            pcall(PerfOpt.AfterShaders)
            task.delay(tween and 1.05 or 0, function()
                ShaderManualLighting.Brightness = ShaderLightingService.Brightness
                ShaderManualLighting.ExposureCompensation = ShaderLightingService.ExposureCompensation
                pcall(PerfOpt.AfterShaders)
            end)
        end
        shaderWarn(report, "Preset " .. tostring(name))
        if ShaderCustomRefresh then
            task.defer(ShaderCustomRefresh)
        end
        if report and report.ok then shaderQueuePersistence() end
        return report
    end

    local function shaderApplySingle(target, property, value)
        if target == "Lighting" and (property == "Brightness" or property == "ExposureCompensation") then

    local function shaderExistingNumber(target, property, fallback)

    local function shaderReadCustomValue(spec, fallback, valueType)
        local override = ShaderCustomOverrides[spec.id]
        if type(override) == valueType then
            return override
        end

        local current = fallback
        if valueType == "boolean" then
            current = shaderExistingBool(spec.target, spec.property, fallback)
        else
            current = shaderExistingNumber(spec.target, spec.property, fallback)
        end

        if type(spec.getValue) == "function" then
            local ok, result = pcall(spec.getValue)
            if ok and type(result) == valueType then
                current = result
            end
        end
        return current
    end

    local function shaderFormatValue(value)
        if math.abs(value) >= 100 then
            return string.format("%.0f", value)
        elseif math.abs(value) >= 10 then
            return string.format("%.1f", value):gsub("%.0$", "")
        end
        return string.format("%.2f", value):gsub("0+$", ""):gsub("%.$", "")
    end

            selected = value == true

            local textColor = selected and Theme.AccentText or Theme.Text
            TweenService:Create(holder, TweenInfo.new(0.16, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {

            }):Play()
            TweenService:Create(glow, TweenInfo.new(0.16, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
                BackgroundColor3 = accentColor,
            }):Play()
            TweenService:Create(badge, TweenInfo.new(0.16, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {

                TextColor3 = textColor,

            badge:SetAttribute("TextEnglish", selected and "✓ ACTIVE" or "APPLY")

        end)

    end
