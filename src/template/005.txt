        local function toHex(c)
            return string.format("#%02X%02X%02X", math.floor(c.R * 255 + 0.5), math.floor(c.G * 255 + 0.5), math.floor(c.B * 255 + 0.5))
        end
        local function fromHex(s)
            s = tostring(s or ""):gsub("#", ""):gsub("%s", "")
            if not s:match("^%x%x%x%x%x%x$") then return nil end
            return Color3.fromRGB(tonumber(s:sub(1, 2), 16), tonumber(s:sub(3, 4), 16), tonumber(s:sub(5, 6), 16))
        end
        function PK.color() return Color3.fromHSV(PK.H, PK.S, PK.V) end
        function PK.refresh()
            local c = PK.color()
            PK.Area.BackgroundColor3 = Color3.fromHSV(PK.H, 1, 1)
            PK.Handle.Position = UDim2.fromScale(PK.S, 1 - PK.V)
            PK.HueHandle.Position = UDim2.fromScale(0.5, PK.H)
            PK.Swatch.BackgroundColor3 = c
            if not PK.Hex:IsFocused() then PK.Hex.Text = toHex(c) end
            if PK.onLive then PK.onLive(c) end
        end
        function PK.set(c)
            local h, s, v = c:ToHSV()
            if s > 0.001 then PK.H = h end
            PK.S, PK.V = s, v
            PK.refresh()
        end
        for i, c in ipairs({
            Color3.fromRGB(255, 255, 255), Color3.fromRGB(255, 82, 82), Color3.fromRGB(255, 160, 60), Color3.fromRGB(255, 214, 70),

            corner(q, 999)
            mk("UIStroke", { Parent = q, Color = WHITE, Thickness = 1, Transparency = 0.5 })
            q.MouseButton1Click:Connect(function() PK.set(c) end)
        end
        function K.pickAt(pos)
            if PK.drag == "SV" then
                local a, s = PK.Area.AbsolutePosition, PK.Area.AbsoluteSize
                if s.X <= 0 or s.Y <= 0 then return end
                PK.S = math.clamp((pos.X - a.X) / s.X, 0, 1)
                PK.V = math.clamp(1 - (pos.Y - a.Y) / s.Y, 0, 1)
            elseif PK.drag == "H" then
                local a, s = PK.Hue.AbsolutePosition, PK.Hue.AbsoluteSize
                if s.Y <= 0 then return end
                PK.H = math.clamp((pos.Y - a.Y) / s.Y, 0, 0.999)
            end
            PK.refresh()
        end
        PK.Area.InputBegan:Connect(function(input) if K.isPress(input) then PK.drag = "SV" K.pickAt(input.Position) end end)
        PK.Hue.InputBegan:Connect(function(input) if K.isPress(input) then PK.drag = "H" K.pickAt(input.Position) end end)
        PK.Hex.FocusLost:Connect(function()
            local c = fromHex(PK.Hex.Text)
            if c then PK.set(c) else PK.refresh() end
        end)
        local function close(ok)
            ov.Visible = false
            PK.drag = nil
            if ok and PK.onApply then PK.onApply(PK.color()) elseif not ok and PK.onCancel then PK.onCancel() end
            PK.onApply, PK.onCancel, PK.onLive = nil, nil, nil
        end
        K.closePicker = close
        cancel.MouseButton1Click:Connect(function() close(false) end)
        apply.MouseButton1Click:Connect(function() close(true) end)

        function K.openPicker(es, en, current, onLive, onApply, onCancel)
            K.L(PK.Title, es, en)
            PK.onLive, PK.onApply, PK.onCancel = nil, onApply, onCancel
            ov.Visible = true
            PK.set(current or WHITE)
            PK.onLive = onLive
        end

        FS.UIKit(K)

        function K.isPress(input)
            return input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch
        end
        function K.getRaw(key)
            local ED = K.ED
            if ED.scope then
                local o = ED.O[ED.scope]
                return o and o[key]
            end
            return ED.G[key]
        end
        function K.setRaw(key, raw)
            local ED = K.ED
            if ED.scope then
                local o = ED.O[ED.scope] or {}
                o[key] = raw
                if next(o) == nil then ED.O[ED.scope] = nil else ED.O[ED.scope] = o end
            else
                ED.G[key] = raw
            end
        end
        function K.get(key)
            local ED = K.ED
            local v = ED.G[key]
            if ED.scope then
                local o = ED.O[ED.scope]
                if o and o[key] ~= nil then v = o[key] end
            end
            if v == "_orig" then return nil end
            return v
        end
        function K.set(key, v)
            if K.ED.scope then
                K.setRaw(key, v == nil and "_orig" or v)
            else
                K.setRaw(key, v)
            end
        end
        function K.resetKeys(keys)
            for _, key in ipairs(keys) do K.setRaw(key, nil) end
            K.RefreshAll()
            K.changed()
        end
