    function GlobalSearch.Refresh()
        GlobalSearch.ClearResults()
        local query = string.lower((GlobalSearch.Box.Text or ""):gsub("^%s+", ""):gsub("%s+$", ""))
        if query == "" then
            GlobalSearch.Empty.Text = GetText("Escribe para buscar una función", "Type to search for a feature")
            GlobalSearch.Empty.Visible = true
            return
        end

        local matches = {}
        for _, entry in ipairs(Window._GlobalOptionRegistry or {}) do
            local nameES = string.lower(tostring(entry.NameSpanish or ""))
            local nameEN = string.lower(tostring(entry.NameEnglish or ""))
            local tabES = string.lower(tostring(entry.Tab and entry.Tab.NameSpanish or ""))
            local tabEN = string.lower(tostring(entry.Tab and entry.Tab.NameEnglish or ""))
            local score = nil
            for _, value in ipairs({nameES, nameEN, tabES, tabEN}) do
                if value ~= "" then
                    local found = string.find(value, query, 1, true)
                    if found then
                        local candidate = found == 1 and 0 or 1
                        score = score == nil and candidate or math.min(score, candidate)
                    end
                end
            end
            if score ~= nil then
                table.insert(matches, {entry = entry, score = score})
            end
        end

        table.sort(matches, function(a, b)
            if a.score ~= b.score then return a.score < b.score end
            return string.lower(tostring(a.entry.NameSpanish or "")) < string.lower(tostring(b.entry.NameSpanish or ""))
        end)

    -- Iconos normales: descarga en segundo plano y conserva el recurso original si falla.
    local YinIcons = { cache = {}, loading = {}, waiting = {} }
    function YinIcons.get(name, callback)
        if YinIcons.cache[name] then callback(YinIcons.cache[name]); return end
        if type(writefile) ~= "function" or type(getcustomasset) ~= "function" then return end
        local queue = YinIcons.waiting[name] or {}
        YinIcons.waiting[name] = queue
        table.insert(queue, callback)
        if YinIcons.loading[name] then return end
        YinIcons.loading[name] = true
        task.spawn(function()
            local ok, asset = pcall(function()
                local path = "YinUI_Icon_v1_" .. name .. ".png"
                if type(isfile) ~= "function" or not isfile(path) then
                    local data = game:HttpGet("https://raw.githubusercontent.com/YinScripts/Yin-UI/refs/heads/main/assets/iconos-yin/" .. name .. ".png")
                    if type(data) ~= "string" or #data < 100 then return nil end
                    writefile(path, data)
                end
                return getcustomasset(path)
            end)
            YinIcons.loading[name] = nil
            local callbacks = YinIcons.waiting[name] or {}
            YinIcons.waiting[name] = nil
            if ok and type(asset) == "string" and asset ~= "" then
                YinIcons.cache[name] = asset
                for _, fn in ipairs(callbacks) do pcall(fn, asset) end
            end
        end)
    end
    function YinIcons.apply(image, name, allowed)
        if not image or not name or (allowed and not allowed()) then return end
        YinIcons.get(name, function(asset)
            if image.Parent and Theme.ThemeName ~= "Navidad" and (not allowed or allowed()) then
                image.Image = asset
                image.ImageColor3 = Color3.new(1, 1, 1)
            end
        end)
    end

    local function keyNameToDisplay(keyName)
        if type(keyName) ~= "string" or keyName == "" then
            return GetText("Sin tecla", "No key")
        end

        local displayName = keyName
        pcall(function()
            local keyCode = Enum.KeyCode[keyName]
            if keyCode then
                local formatted = UserInputService:GetStringForKeyCode(
                    keyCode,
                    Enum.KeyCodeStringFormat.Short
                )
                if type(formatted) == "string" and formatted ~= "" then
                    displayName = formatted
                end
            end
        end)
        return displayName
    end

    Window.RegisterKeybind = function(id, button, activate)
        if type(id) ~= "string" or id == "" or not button or type(activate) ~= "function" then
            return nil
        end

        local entry = Window.Keybinds[id] or {}
        entry.Button = button
        entry.Activate = activate
        entry.KeyName = SavedConfig.Keybinds[id]
        entry.Capturing = false

        entry.Update = function()
            updateKeybindButton(entry)
        end

        entry.SetKey = function(keyName)
            if type(keyName) ~= "string" or keyName == "" then return end

            for otherId, other in pairs(Window.Keybinds) do
                if otherId ~= id and other.KeyName == keyName then
                    other.KeyName = nil
                    SavedConfig.Keybinds[otherId] = nil
                    if other.Update then other.Update() end
                end
            end

            entry.KeyName = keyName
            SavedConfig.Keybinds[id] = keyName
            entry.Capturing = false
            if Window._keybindCapture == entry then
                Window._keybindCapture = nil
            end
            updateKeybindButton(entry)
            SaveConfig()
        end

        entry.Clear = function()
            entry.KeyName = nil
            SavedConfig.Keybinds[id] = nil
            entry.Capturing = false
            if Window._keybindCapture == entry then
                Window._keybindCapture = nil
            end
            updateKeybindButton(entry)
            SaveConfig()
        end

        Window.Keybinds[id] = entry
        updateKeybindButton(entry)
        return entry
    end
