            Kit.New("Frame", {Size = UDim2.new(1, 0, 0, 16), Position = UDim2.new(0, 0, 1, -16), BorderSizePixel = 0,
                BackgroundTransparency = o.HeaderTransparency or 0.15}, head, {bg = "Secondary"})

                Image = "rbxthumb://type=AvatarHeadShot&id=" .. tostring(o.Avatar) .. "&w=150&h=150"}, head, {bg = "Secondary"})
            Kit.Corner(av, 20)
            Kit.Stroke(av, "Accent", 2, 0.2)
            x = 60
        end
        local right = 12
        local function headButton(text)
            local b = Btn(head, {AnchorPoint = Vector2.new(1, 0.5), Position = UDim2.new(1, -right, 0.5, 0), Size = UDim2.fromOffset(28, 28),

            Kit.Corner(b, 14)
            right = right + 34

        end

        end

            head.Position = UDim2.new()
            filler.LayoutOrder, body.LayoutOrder = 1, 2
        end
        W.Body = body
        setmetatable(W, {__index = function(t, k)
            local f = Widgets[k]
            if f then return function(self, ...) return f(t._widgets, ...) end end
            return nil
        end})
        W._widgets = container(body, o.Gap or 8)
        function W:SetTitle(t) W.TitleLabel.Text = t end
        function W:SetSubtitle(t) if W.SubtitleLabel then W.SubtitleLabel.Text = t end end
        function W:SetInfo(t) if W.InfoLabel then W.InfoLabel.Text = t end end

            if not W._pinned then

                    auto and frame or body)
                if auto then
                    top.LayoutOrder = 2
                    body.LayoutOrder = 3
                    Kit.Pad(top, 8, pad, 0, pad)
                end
                Kit.List(top, o.Gap or 8)
                W._pinned = container(top, o.Gap or 8)
            end
            return W._pinned
        end
        function W:SetMinimized(m)
            W.Minimized = m
            body.Visible = not m
            if W.MinButton then W.MinButton.Text = m and "+" or "-" end
            if not auto then
                frame.Size = m and UDim2.new(frame.Size.X.Scale, frame.Size.X.Offset, 0, headH) or W._fullSize
            end
        end
        W._fullSize = frame.Size
        function W:Show()
            gui.Enabled = true
            W.Scale.Scale = 0.9
            Kit.Tween(W.Scale, 0.3, {Scale = 1}, Enum.EasingStyle.Back)
            if W.Shade then Kit.Tween(W.Shade, 0.25, {BackgroundTransparency = 0.45}) end
        end
        function W:Hide() gui.Enabled = false end
        function W:Close()
            if W.Closed then return end
            if o.OnClose then
                local keep = o.OnClose()
                if keep == true then return end
            end
            W:Destroy()
        end
        function W:Destroy()
            if W.Closed then return end
            W.Closed = true
            if W.Shade then
                Kit.Tween(W.Shade, 0.2, {BackgroundTransparency = 1})
                local tw = Kit.Tween(W.Scale, 0.2, {Scale = 0.88}, Enum.EasingStyle.Quad, Enum.EasingDirection.In)
                tw.Completed:Wait()
            end
            if gui.Parent then gui:Destroy() end
        end
        W:Show()
        return W
    end

-- ===== PIEZAS DE PÁGINA (contenido dentro de las pestañas) =====
-- Encabezado con ícono, tarjeta, texto, campo numerado, botón grande y línea de estado: las arma la librería con el
-- tema activo (atributos de rol) y el script solo pasa los textos. Idioma: si se pasan Es/En quedan guardados en los
-- atributos TextSpanish/TextEnglish (y PlaceholderSpanish/PlaceholderEnglish) para que el script los traduzca.
do
    local Kit = YinYang.Kit
    local Page = {}
    Kit.Page = Page
    -- instancia simple (sin tocar AutoLocalize / Selectable, igual que las piezas que antes armaba el script)
    local function new(class, props, parent)
        local o = Instance.new(class)
        for k, v in pairs(props) do o[k] = v end
        if parent then o.Parent = parent end
        return o
    end

    local function lang(obj, es, en, placeholder)
        if es == nil or en == nil then return end
        obj:SetAttribute(placeholder and "PlaceholderSpanish" or "TextSpanish", es)
        obj:SetAttribute(placeholder and "PlaceholderEnglish" or "TextEnglish", en)
    end
    -- borde fino con el color "Stroke" del tema
    function Page.Stroke(parent, transparency)
        local s = new("UIStroke", {Color = Kit.Palette().Stroke, Thickness = 1, Transparency = transparency or 0.55}, nil)
        pcall(function() s.ZIndex = 12 end)
        s.Parent = parent
        s:SetAttribute("ThemeRole", "Stroke")
        return s
    end
    -- color de estado que se lee en temas claros (lo oscurece a la mitad si el fondo es claro)
    function Page.AdaptStatus(color)
        local bg = Kit.Palette().Background
        if 0.299 * bg.R + 0.587 * bg.G + 0.114 * bg.B > 0.55 then
            return Color3.new(color.R * 0.5, color.G * 0.5, color.B * 0.5)
        end
        return color
    end

    -- tarjeta: o = {Name, Height, Transparency, LayoutOrder, StrokeTransparency}; devuelve tarjeta y borde
    function Page.Card(parent, o)
        o = o or {}
        local card = new("Frame", {Name = o.Name or "Card", Size = UDim2.new(1, 0, 0, o.Height or 50), BackgroundColor3 = Kit.Palette().Secondary,
            BackgroundTransparency = o.Transparency or 0.78, BorderSizePixel = 0, ZIndex = 10}, nil)
        if o.LayoutOrder then card.LayoutOrder = o.LayoutOrder end
        card.Parent = parent
        card:SetAttribute("ThemeRole", "Secondary")
        Kit.Corner(card, 10)
        local stroke = Page.Stroke(card, o.StrokeTransparency or 0.55)
        return card, stroke
    end

    -- texto: o = {Text, Es, En, TextSize, Font, Role, Position, Size}
    function Page.Label(parent, o)
        local role = o.Role or "Text"
        local label = new("TextLabel", {BackgroundTransparency = 1, Position = o.Position or UDim2.new(), Size = o.Size or UDim2.new(1, 0, 0, 16),
            Text = o.Text or "", TextColor3 = Kit.Palette()[role], TextSize = o.TextSize or 12, Font = o.Font or Enum.Font.GothamMedium,
            TextWrapped = true, TextXAlignment = Enum.TextXAlignment.Left, TextYAlignment = Enum.TextYAlignment.Top, ZIndex = 12}, parent)
        label:SetAttribute("ThemeTextRole", role)
        label:SetAttribute("IgnoreTextEffect", true)
        lang(label, o.Es, o.En)
        return label
    end

    -- encabezado de pestaña: tarjeta de 78 con barrita de acento, caja con ícono, título, ✦ y descripción
    -- o = {Name, BarName, DescName, LayoutOrder, Icon, IconFn (ícono que cambia con el tema), Title, TitleEs, TitleEn,
    --      Desc, DescEs, DescEn, PlainTitle (título que sí toma los efectos de texto de la librería)}
    -- devuelve la tarjeta y sus piezas {Bar, IconBox, Icon, Title, Spark, Desc}
    function Page.Header(paren
