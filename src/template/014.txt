        setByMe = setmetatable({}, {__mode = "k"}), hooked = setmetatable({}, {__mode = "k"}),
        props = {TextLabel = {"Text"}, TextButton = {"Text"}, TextBox = {"PlaceholderText"}},

        local Tr = Chat.tr
        Tr.base = function(l) return string.lower(tostring(l or "")):match("^(%a+)") or "" end
        Tr.detect = function()
            local loc = ""
            pcall(function() loc = tostring(game:GetService("LocalizationService").RobloxLocaleId or "") end)
            if loc == "" then pcall(function() loc = tostring(LocalPlayer.LocaleId or "") end) end
            if loc == "" then pcall(function() loc = tostring(game:GetService("LocalizationService").SystemLocaleId or "") end) end
            loc = string.lower(loc ~= "" and loc or "es-es"):gsub("_", "-")

            if lang == "zh" then lang = (loc:find("tw") or loc:find("hk") or loc:find("hant")) and "zh-TW" or "zh-CN" end
            return lang, loc
        end
        Tr.lang, Tr.locale = Tr.detect()
        Tr.uiOn = Tr.base(Tr.lang) ~= "es"
        Tr.file = "YinYang_ChatLang_" .. Tr.lang .. ".json"
        pcall(function()
            if Tr.uiOn and isfile(Tr.file) then
                local d = HttpService:JSONDecode(readfile(Tr.file))
                if type(d) == "table" then
                    for k, v in pairs(d) do
                        if type(k) == "string" and type(v) == "string" then Tr.cache[k] = v end
                    end
                end
            end
        end)
        Tr.save = function()
            if Tr.saving then return end
            Tr.saving = true
            task.delay(3, function()
                Tr.saving = false
                pcall(function() writefile(Tr.file, HttpService:JSONEncode(Tr.cache)) end)
            end)

        Tr.google = function(text, tl)
            local ok, res = pcall(function()
                return request({
                    Url = "https://translate.googleapis.com/translate_a/single?client=gtx&sl=auto&tl=" .. HttpService:UrlEncode(tl)
                        .. "&dt=t&q=" .. HttpService:UrlEncode(text),
                    Method = "GET",
                })
            end)
            if not ok or type(res) ~= "table" or type(res.Body) ~= "string" or res.Body == "" then return nil end
            local okD, data = pcall(HttpService.JSONDecode, HttpService, res.Body)
            if not okD or type(data) ~= "table" or type(data[1]) ~= "table" then return nil end
            local out = {}
            for _, seg in ipairs(data[1]) do
                if type(seg) == "table" and type(seg[1]) == "string" then out[#out + 1] = seg[1] end
            end
            if #out == 0 then return nil end
            local from = type(data[3]) == "string" and data[3] or nil
            if not from then
                for i, v in pairs(data) do
                    if i ~= 1 and type(v) == "string" then from = v break end
                end
            end
            return table.concat(out), from
        end
        Tr.roblox = function(text, tl)
            local ok, out = pcall(function()
                local r = TextService:FilterAndTranslateStringAsync(text, LocalPlayer.UserId, {tl})
                local t = r:GetTranslations()
                if type(t) ~= "table" then return nil end
                local v = t[tl] or t[Tr.locale]
                if v == nil then for _, x in pairs(t) do v = x break end end
                return type(v) == "string" and v or nil
            end)
            return ok and out or nil
        end
        Tr.probe = function()
            if Tr.robloxOk ~= nil then return Tr.robloxOk end
            local sample = Tr.base(Tr.lang) == "es" and "Hello, how are you today?" or "Hola, ¿cómo estás hoy?"
            local out = Tr.roblox(sample, Tr.lang)
            Tr.robloxOk = type(out) == "string" and out ~= "" and out ~= sample
            return Tr.robloxOk
        end
        Tr.one = function(text)
            if Tr.probe() then
                local out = Tr.roblox(text, Tr.lang)
                if out and out ~= "" then return out, nil end
            end
            return Tr.google(text, Tr.lang)
        end
        Tr.trim = function(s) return (tostring(s or ""):match("^%s*(.-)%s*$")) end
        Tr.batch = function(list)
            local res = {}
            if not Tr.probe() and #list > 1 then
                local joined = Tr.google(table.concat(list, "\n"), Tr.lang)
                if not joined then return res end
                do
                    local parts = {}
                    for line in (joined .. "\n"):gmatch("(.-)\n") do parts[#parts + 1] = Tr.trim(line) end
                    if #parts == #list then
                        for i, s in ipairs(list) do res[s] = parts[i] end
                        return res
                    end
                end
            end
            for i, s in ipairs(list) do
                res[s] = (Tr.one(s))
                if i < #list then task.wait(0.1) end
            end
            return res
        end
        Tr.want = function(key, cb)
            local c = Tr.cache[key]
            if c then cb(c) return end
            local w = Tr.waiters[key]
            if w then table.insert(w, cb) return end
            Tr.waiters[key] = {cb}
            table.insert(Tr.queue, key)
            Tr.kick()
        end
        Tr.kick = function()
            if Tr.running then return end
            Tr.running = true
            task.spawn(function()
                task.wait(0.3)
                while #Tr.queue > 0 and ScreenGui.Parent do
                    local list, size = {}, 0
                    while #Tr.queue > 0 and #list < 25 and (#list == 0 or size + #Tr.queue[1] < 1200) do
                        local k = table.remove(Tr.queue, 1)
                        list[#list + 1] = k
                        size += #k + 1
                    end
                    local res = Tr.batch(list)
                    local anyOk = false
                    for _, k in ipairs(list) do
                        local v = res[k]
                        if type(v) == "string" and v ~= "" then
                            Tr.cache[k] = v
                            anyOk = true
                        end
                        Tr.tries[k] = (Tr.tries[k] or 0) + 1
                        if Tr.cache[k] or Tr.tries[k] >= 3 then
                            local w = Tr.waiters[k]
                            Tr.waiters[k] = nil
                            for _, cb in ipairs(w or {}) do pcall(cb, Tr.cache[k]) end
                        else
                            table.insert(Tr.queue, k)
                        end
                    end
                    if anyOk then Tr.save() else task.wait(8) end
                    task.wait(0.2)
                end
                Tr.running = false
            end)
        end
        Tr.convert = function(src, cb)
            local pieces, pos = {}, 1
            while pos <= #src do
                local a, b = src:find("<[^<>]->", pos)
                if not a then
                    pieces[#pieces + 1] = {t = src:sub(pos)}
                    break
                end
                if a > pos then pieces[#pieces + 1] = {t = src:sub(pos, a - 1)} end
                pieces[#pieces + 1] = {tag = src:sub(a, b)}
                pos = b + 1
            end
            local pending = 1
            local function done()
                pending -= 1
                if pending > 0 then return end
                local out = {}
                for _, p in ipairs(pieces) do out[#out + 1] = p.tag or p.out or p.t end
                cb(table.concat(out))
            end
            for _, p in ipairs(pieces) do
                if p.t then
                    local lead, core, trail = p.t:match("^(%s*)(.-)(%s*)$")
                    if core:find("%a") then
                        local nums = {}
                        local key = core:gsub("%d+", function(n)
                            nums[#nums + 1] = n
                            return "0"
                        end)
                        pending += 1
                        Tr.want(key, function(v)
                            if type(v) == "string" then
                                local i = 0
                                local filled = v:gsub("%d+", function()
                                    i += 1
                                    return nums[i]
                                end)
                                if i == #nums then p.out = lead .. filled .. trail end
                            end
                            done()
                        end)
                    end
                end
            end
            done()
        end
        Tr.apply = function(obj, prop)
            if not obj.Parent or obj:GetAttribute("NoTr") then return end
            local cur = obj[prop]
            if type(cur) ~= "string" or cur == "" or not cur:find("%a") then return end
            local mine = Tr.setByMe[obj]
            if mine and mine[prop] == cur then return end
            Tr.convert(cur, function(out)
                if out ~= cur and obj[prop] == cur and obj.Parent and not obj:GetAttribute("NoTr") then
                    Tr.setByMe[obj] = Tr.setByMe[obj] or {}
                    Tr.setByMe[obj][prop] = out
                    obj[prop] = out
                end
            end)
        end
        Tr.hook = function(obj)
            local props = Tr.props[obj.ClassName]
            if not props or Tr.hooked[obj] then return end
            Tr.hooked[obj] = true
            for _, prop in ipairs(props) do
                obj:GetPropertyChangedSignal(prop):Connect(function() Tr.apply(obj, prop) end)
                Tr.apply(obj, prop)
            end
        end
        Tr.watch = function(root)
            for _, d in ipairs(root:GetDescendants()) do Tr.hook(d) end
            root.DescendantAdded:Connect(function(d)
                task.defer(function()
                    if d.Parent then Tr.hook(d) end
                end)
            end)
        end
        Tr.msgFor = function(src, cb)
            local c = Tr.msgCache[src]
            if c ~= nil then cb(c or nil) return end
            local w = Tr.msgWait[src]
            if w then table.insert(w, cb) return end
            Tr.msgWait[src] = {cb}
            table.insert(Tr.msgQueue, src)
            if Tr.msgRunning then return end
            Tr.msgRunning = true
            task.spawn(function()
                while #Tr.msgQueue > 0 and ScreenGui.Parent do
                    local s = table.remove(Tr.msgQueue, 1)
                    local out, from = Tr.one(s)
                    local res = nil
                    if out ~= nil then
                        res = false
                        local same = string.lower((Tr.trim(out):gsub("%s+", " "))) == string.lower((Tr.trim(s):gsub("%s+", " ")))
                        if out ~= "" and not same and (from == nil or Tr.base(from) ~= Tr.base(Tr.lang)) then res = out end
                        Tr.msgCache[s] = res
                    end
                    local list = Tr.msgWait[s]
                    Tr.msgWait[s] = nil
                    for _, f in ipairs(list or {}) do pcall(f, res or nil) end
                    task.wait(out == nil and 3 or 0.15)
                end
                Tr.msgRunning = false
            end)
        end
        Tr.message = function(meta)
            if type(meta) ~= "table" or meta.isMe or meta.trReq or not meta.trBody or type(meta.trSrc) ~= "string" then return end
            if not meta.trSrc:find("%a") then return end
            meta.trReq = true
            Tr.msgFor(meta.trSrc, function(out)
                if not out or not meta.row or not meta.row.Parent or messageRows[meta.key] ~= meta then return end
                local nearBottom = Chat.isNearBottom()
                meta.trOut = out
                meta.trShown = true
                meta.trBody(out)
                local btn = mk("TextButton", {
                    Size = UDim2.fromOffset(0, 18), AutomaticSize = Enum.AutomaticSize.X, BackgroundTransparency = 1,
                    Text = "Traducido · Ver original", TextColor3 = THEME.DIM, Font = Enum.Font.GothamBold, TextSize = 12,
                    AutoButtonColor = false, LayoutOrder = 5,
                }, meta.content)
                mk("UIPadding", {PaddingLeft = UDim.new(0, 6), PaddingRight = UDim.new(0, 6)}, btn)
                meta.trBtn = btn
                btn.MouseButton1Click:Connect(function()
                    meta.trShown = not meta.trShown
                    meta.trBody(meta.trShown and meta.trOut or meta.trSrc)
                    btn.Text = meta.trShown and "Traducido · Ver original" or "Ver traducción"
                end)
                if nearBottom then task.defer(scrollBottom) end
            end)
        end
        if Tr.uiOn then
            Tr.watch(ScreenGui)
            Tr.watch(Surface)
        end
        for _, meta in pairs(messageRows) do Tr.message(meta) end
    end
    Chat.applyView()
    Chat.refreshHome(true)
    task.spawn(function() pcall(Chat.refreshPresence) end)
    Chat.layoutHost()
    return {
        Chat = Chat,
        SetHeaderInset = function(inset)
            inset = math.max(0, tonumber(inset) or 0)
            Header.Position = UDim2.fromOffset(inset, 0)
            Header.Size = UDim2.new(1, -inset, 0, 58)
        end,
        GetSideNotifs = function() return Chat.settings.SideNotifs ~= false end,
        SetSideNotifs = function(v)
            Chat.settings.SideNotifs = v and true or false
            Chat.saveSettings()
            if not v and UI.SideHolder then
                for _, c in ipairs(UI.SideHolder:GetChildren()) do
                    if c:IsA("GuiObject") then c:Destroy() end

        Destroy = function()

    local resolveLogoClickSound
    do
        local resolved = {}
        resolveLogoClickSound = function(iconName, clickS, allowDownload)
            local url = clickS and clickS.Url
            if type(url) ~= "string" or url == "" then return nil end

            local cacheName = clickS.CacheName
            if type(cacheName) ~= "string" or cacheName == "" then
                cacheName = "YinYang_IconClick_" .. tostring(iconName) .. ".mp3"
            end

            if resolved[cacheName] then return resolved[cacheName] end
            if not getcustomasset then return nil end

            local exists = false
            if isfile then
                local okFile, resultFile = pcall(function() return isfile(cacheName) end)
                exists = okFile and resultFile == true
            end

            if not exists then
                if not allowDownload or not writefile then return nil end

                local okDownload, data = pcall(function() return game:HttpGet(url, true) end)
                if not okDownload or type(data) ~= "string" or #data < 10 then
                    warn("[YinYang Icons] Falló la descarga del sonido (" .. url .. "): " .. tostring(data))
                    return nil
                end

                local okWrite = pcall(function() writefile(cacheName, data) end)
                if not okWrite then
                    warn("[YinYang Icons] Falló writefile para el sonido: " .. cacheName)
                    return nil
                end
            end

            local okAsset, asset = pcall(function() return getcustomasset(cacheName) end)
            if okAsset and type(asset) == "string" and asset ~= "" then
                resolved[cacheName] = asset
                return asset
            end
            return nil
        end
    end

    local function ApplyLogoIcon(iconName)
        if logoRenderConn then logoRenderConn:Disconnect(); logoRenderConn = nil end
        if logoIdleConn   then logoIdleConn:Disconnect();   logoIdleConn   = nil end

        for _, entry in ipairs(logoLayers) do
            if entry.label and entry.label.Parent then
                entry.label:Destroy()
            end
        end
        logoLayers = {}

        stopGlowTweens(tostring(ToggleButton))
        local _cs = ToggleButton:FindFirstChildOfClass("UIStroke")
        if _cs then
            local _cg = _cs:FindFirstChildOfClass("UIGradient")
            if _cg then _cg:Destroy() end
            _cs.Transparency = 0
            _cs.Color        = Theme.Accent
            _cs.Thickness    = 1.5
        end
        ToggleButton:SetAttribute("IsCapsule", false)
        ToggleButton.Size             = UDim2.fromOffset(42, 42)
        ToggleButton.BackgroundColor3 = Theme.Accent
        ToggleButton.BackgroundTransparency = 0
        do
            local _resetCorner = ToggleButton:FindFirstChildOfClass("UICorner")
            if _resetCorner then _resetCorner.CornerRadius = UDim.new(0, 999) end
        end

        local iconData = IconCatalog.Icons[iconName]
            or IconCatalog.Icons["ClassicYinYang"]
        if not iconData then return end

        if iconData.ClickSound and type(iconData.ClickSound.Url) == "string" and iconData.ClickSound.Url ~= "" then
            task.spawn(resolveLogoClickSound, iconName, iconData.ClickSound, true)
        end

        if iconData.Type == "Capsule" then
            ToggleButton:SetAttribute("IsCapsule", true)
            ToggleButton.Size             = UDim2.fromOffset(185, 42)
            ToggleButton.BackgroundColor3 = Color3.fromRGB(18, 18, 24)
            ToggleButton.BackgroundTransparency = 0.15

            do
                local _capCorner = ToggleButton:FindFirstChildOfClass("UICorner")
                if _capCorner then _capCorner.CornerRadius = UDim.new(0, 14) end
            end

            local arrowImg = mk("ImageLabel", {
                Parent                 = ToggleButton,
                Size                   = UDim2.fromOffset(26, 26),
                Position               = UDim2.new(0, 10, 0.5, -13),
                BackgroundTransparency = 1,
                Image                  = "rbxassetid://89220810765334",
                ImageColor3            = Color3.fromRGB(255, 255, 255),
                ZIndex                 = 31,
            })
            table.insert(logoLayers, { label = arrowImg, def = {}, state = {} })

            local divider = mk("Frame", {
                Parent           = ToggleButton,
                Size             = UDim2.fromOffset(1, 20),
                Position         = UDim2.new(0, 42, 0.5, -10),
                BackgroundColor3 = Color3.fromRGB(80, 80, 100),
                BorderSizePixel  = 0,
                ZIndex           = 31,
            })
            table.insert(logoLayers, { label = divider, def = {}, state = {} })

            local textLbl = mk("TextLabel", {
                Parent                 = ToggleButton,
                Size                   = UDim2.new(1, -50, 1, 0),
                Position               = UDim2.fromOffset(48, 0),
                BackgroundTransparency = 1,
                Text                   = "Open YinYang",
                TextColor3             = Color3.fromRGB(255, 255, 255),
                Font                   = Enum.Font.GothamBold,
                TextSize               = 16,
                TextXAlignment         = Enum.TextXAlignment.Center,
                ZIndex                 = 31,
            })
            table.insert(logoLayers, { label = textLbl, def = {}, state = {} })

            do
                local _key = tostring(ToggleButton)
                stopGlowTweens(_key)

                local _capStroke = ToggleButton:FindFirstChildOfClass("UIStroke")
                if _capStroke then
                    local _oldGrad = _capStroke:FindFirstChildOfClass("UIGradient")
                    if _oldGrad then _oldGrad:Destroy() end
                    _capStroke.Thickness    = 3
                    _capStroke.Color        = Color3.fromRGB(255, 255, 255)
                    _capStroke.Transparency = 0
                    _capStroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
                    _capStroke.LineJoinMode    = Enum.LineJoinMode.Round

                    local _capsulePulse = TweenService:Create(
                        _capStroke,
                        TweenInfo.new(1.6, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut, -1, true),
                        { Transparency = 0.15 }
                    )
                    _capsulePulse:Play()

                    local _bandA = mk("Frame", {
                        Parent = ToggleButton,
                        Size = UDim2.new(1, 0, 1, 0),
                        BackgroundTransparency = 1,
                        ZIndex = 40,
                    })
                    corner(_bandA, 14)
                    local _bandAStroke = mk("UIStroke", {
                        Color = Color3.fromRGB(0, 0, 0),
                        Thickness = 3,
                        Transparency = 0,
                        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
                        LineJoinMode = Enum.LineJoinMode.Round,
                    }, _bandA)
                    local _gradA = Instance.new("UIGradient")
                    _gradA.Color = ColorSequence.new(Color3.fromRGB(0, 0, 0))
                    _gradA.Transparency = NumberSequence.new({
                        NumberSequenceKeypoint.new(0.00, 1),
                        NumberSequenceKeypoint.new(0.35, 1),
                        NumberSequenceKeypoint.new(0.50, 0.15),
                        NumberSequenceKeypoint.new(0.65, 1),
                        NumberSequenceKeypoint.new(1.00, 1),
                    })
                    _gradA.Offset = Vector2.new(-1.5, 0)
                    _gradA.Parent = _bandAStroke
                    local _sweepA = TweenService:Create(
                        _gradA,
                        TweenInfo.new(2.2, Enum.EasingStyle.Linear, Enum.EasingDirection.InOut, -1, false),
                        { Offset = Vector2.new(1.5, 0) }
                    )
                    _sweepA:Play()
                    table.insert(logoLayers, { label = _bandA, def = {}, state = {} })

                    local _bandB = mk("Frame", {
                        Parent = ToggleButton,
                        Size = UDim2.new(1, 0, 1, 0),
                        BackgroundTransparency = 1,
                        ZIndex = 40,
                    })
                    corner(_bandB, 14)
                    local _bandBStroke = mk("UIStroke", {
                        Color = Color3.fromRGB(0, 0, 0),
                        Thickness = 3,
                        Transparency = 0,
                        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
                        LineJoinMode = Enum.LineJoinMode.Round,
                    }, _bandB)
                    local _gradB = Instance.new("UIGradient")
                    _gradB.Color = ColorSequence.new(Color3.fromRGB(0, 0, 0))
                    _gradB.Transparency = NumberSequence.new({
                        NumberSequenceKeypoint.new(0.00, 1),
                        NumberSequenceKeypoint.new(0.35, 1),
                        NumberSequenceKeypoint.new(0.50, 0.15),
                        NumberSequenceKeypoint.new(0.65, 1),
                        NumberSequenceKeypoint.new(1.00, 1),
                    })
                    _gradB.Offset = Vector2.new(1.5, 0)
                    _gradB.Parent = _bandBStroke
                    local _sweepB = TweenService:Create(
                        _gradB,
                        TweenInfo.new(2.2, Enum.EasingStyle.Linear, Enum.EasingDirection.InOut, -1, false),
                        { Offset = Vector2.new(-1.5, 0) }
                    )
                    _sweepB:Play()
                    table.insert(logoLayers, { label = _bandB, def = {}, state = {} })

                    ActiveGlowTweens[_key] = { _capsulePulse, _sweepA, _sweepB }
                end
            end
            return
        end

        if type(iconData.Build) == "function" then
            local _bs = ToggleButton:FindFirstChildOfClass("UIStroke")
            ToggleButton.Size = UDim2.fromOffset(56, 56)
            ToggleButton.BackgroundTransparency = 1
            if _bs then _bs.Transparency = 1 end

            local _holder = mk("Frame", {
                Parent                 = ToggleButton,
                Size                   = UDim2.fromScale(1, 1),
                BackgroundTransparency = 1,
                ZIndex                 = 31,
            })
            local _bOk, _bRes = pcall(iconData.Build, { Parent = _holder, Button = ToggleButton })
            if _bOk then
                table.insert(logoLayers, {
                    label   = _holder,
                    def     = {},
                    state   = {},
                    onClick = (type(_bRes) == "table") and _bRes.OnClick or nil,
                })
                return
            end

            print("[YinYang Icons] ⚠️ Build falló, se usa Layers: " .. tostring(_bRes))
            _holder:Destroy()
            ToggleButton.Size = UDim2.fromOffset(42, 42)
            ToggleButton.BackgroundTransparency = 0
            if _bs then _bs.Transparency = 0 end
        end

        if not iconData.Layers then return end

        -- NoBackground = true (catalogo de iconos): solo se ve la imagen, sin el circulo de color
        -- del tema ni el borde del boton. Ambos se restauran al inicio de ApplyLogoIcon al cambiar de icono.
        if iconData.NoBackground == true then
            local _nbStroke = ToggleButton:FindFirstChildOfClass("UIStroke")
            ToggleButton.BackgroundTransparency = 1
            if _nbStroke then _nbStroke.Transparency = 1 end
        end
        -- BgColor / BorderColor (logos propios del Estudio de logos): colores fijos en vez del Accent del tema.
        -- Los íconos del catálogo no los traen, así que para ellos no cambia nada.
        if iconData.NoBackground ~= true then
            if typeof(iconData.BgColor) == "Color3" then
                ToggleButton.BackgroundColor3 = iconData.BgColor
            end
            if typeof(iconData.BorderColor) == "Color3" then
                local _bcStroke = ToggleButton:FindFirstChildOfClass("UIStroke")
                if _bcStroke then _bcStroke.Color = iconData.BorderColor end
            end
        end

        for i, layerDef in ipairs(iconData.Layers) do
            local sc = layerDef.Scale or 1.6
            local off = -((sc - 1) / 2)
            local label
            local spriteState = nil

            if type(layerDef.SpriteSheet) == "table" then
                label, spriteState = createSpriteSheetLabels(
                    ToggleButton,
                    layerDef.SpriteSheet,
                    UDim2.new(sc, 0, sc, 0),
                    UDim2.new(off, 0, off, 0),
                    30 + i
                )
            else
                label = mk("ImageLabel", {
                    Parent               = ToggleButton,
                    Size                 = UDim2.new(sc, 0, sc, 0),
                    Position             = UDim2.new(off, 0, off, 0),
                    BackgroundTransparency = 1,
                    Image                = layerDef.Image or "",
                    ZIndex                = 30 + i,
                })
            end

            if not label then
                continue
            end

            -- Color / Transparency por capa (logos propios); las capas del catálogo no los traen
            if label:IsA("ImageLabel") then
                if typeof(layerDef.Color) == "Color3" then
                    label.ImageColor3 = layerDef.Color
                end
                if tonumber(layerDef.Transparency) then
                    label.ImageTransparency = math.clamp(tonumber(layerDef.Transparency), 0, 1)
                end
            end

            mk("UIAspectRatioConstraint", { AspectRatio = 1 }, label)

            local state = {
                angle       = 0,
                speed       = layerDef.Speed or 60,
                tSpeed      = layerDef.Speed or 60,
                time        = 0,
                turboEndTime = 0,
                turboSpeed   = layerDef.TurboSpeed or 350,
                sprite      = spriteState,
            }
            table.insert(logoLayers, { label = label, def = layerDef, state = state })
        end

        logoRenderConn = game:GetService("RunService").RenderStepped:Connect(function(dt)
            if PerfOpt.Flags.NoLogoAnim then return end
            for _, entry in ipairs(logoLayers) do
                local lbl = entry.label
                local def = entry.def
                local st  = entry.state
                if not lbl or not lbl.Parent then continue end

                local mv  = def.Movement  or "None"
                local spd = def.Speed     or 60
                local amt = def.Amount    or 0
                local dir = def.Direction or 1

                st.time = st.time + dt

                if st.sprite then
                    local sprite = st.sprite
                    sprite.Acc = sprite.Acc + dt
                    local step = 1 / sprite.Config.FPS
                    if sprite.Acc > 0.25 then
                        sprite.Acc = 0.25
                    end
                    local moved = false
                    while sprite.Acc >= step do
                        sprite.Acc = sprite.Acc - step
                        sprite.Frame = (sprite.Frame + 1) % sprite.Config.TotalFrames
                        moved = true
                    end
                    if moved then
                        updateSpriteSheetFrame(sprite, sprite.Frame)
                    end
                end

                if mv == "Spin" then
                    local now    = os.clock()
                    local inTurbo = st.turboEndTime > 0 and now < st.turboEndTime

                    if inTurbo then
                        st.speed  = st.turboSpeed
                        st.tSpeed = st.turboSpeed
                    else
                        if (def.Style or "Constant") == "Organic" then
                            st.tSpeed = spd + math.sin(st.time * 0.8) * (spd * 0.33)
                            st.speed  = lerpN(st.speed, st.tSpeed, 0.08)
                        else
                            st.speed = spd
                        end
                    end
                    st.angle = st.angle + st.speed * dir * dt
                    lbl.Rotation = st.angle

                elseif mv == "Pulse" then
                    local sc    = def.Scale or 1.6
                    local off   = -((sc - 1) / 2)
                    local scale = 1 + math.sin(st.time * (2 * math.pi / spd)) * ((amt ~= 0 and amt or 12) / 100)
                    lbl.Size     = UDim2.new(sc * scale, 0, sc * scale, 0)
                    lbl.Position = UDim2.new(off * scale, 0, off * scale, 0)

                elseif mv == "Orbit" then
                    local sc     = def.Scale or 1.6
                    local off    = -((sc - 1) / 2)
                    local radius = amt ~= 0 and amt or 6
                    local angle  = st.time * math.rad(spd) * dir
                    lbl.Position = UDim2.new(off, math.cos(angle) * radius, off, math.sin(angle) * radius)

                elseif m
