-- ===== GUÍA (tutorial navegable dentro de una pestaña) =====
-- Tarjeta de inicio con buscador, chips por pestaña y tarjetas que se abren al tocarlas. El script pasa solo el
-- contenido (Data) y los textos; la librería arma todo con el tema activo. Los textos se aclaran u oscurecen solos si
-- con algún tema quedan con poco contraste, y se repintan al volver a la pestaña.
-- Kit.Page.Guide(page, o): o = {
--   Data = { {Key, Es, En, Icon, Sum = {es, en}, Sections = { {tituloEs, tituloEn, { item, ... }}, ... }}, ... }
--        item = {n = {es, en}, s = {es, en}, h = {es, en} (opcional), sl = { {clave, es, en}, ... } (opcional), t = {etiqueta, ...}}
--   Tags = {CLAVE = {es, en}}, TagColors = {CLAVE = Color3} (las demás usan el acento),
--   T = function(es, en) (texto en el idioma actual), Badge = "?", Title = {es, en}, Subtitle = {es, en},
--   Placeholder = {es, en}, NotFound = function(busqueda) return es, en end, Start = clave de la pestaña inicial }
do
    local Kit = YinYang.Kit
    local Page = Kit.Page
    local function new(class, props, parent)
        local o = Instance.new(class)
        for k, v in pairs(props) do o[k] = v end
        if parent then o.Parent = parent end
        return o
    end
    local function round(inst, r) return new("UICorner", {CornerRadius = UDim.new(0, r)}, inst) end
    local function lum(c)
        local function ch(v) return v <= 0.03928 and v / 12.92 or ((v + 0.055) / 1.055) ^ 2.4 end
        return 0.2126 * ch(c.R) + 0.7152 * ch(c.G) + 0.0722 * ch(c.B)
    end
    local function contrast(a, b)
        local la, lb = lum(a), lum(b)
        if la < lb then la, lb = lb, la end
        return (la + 0.05) / (lb + 0.05)
    end
    -- acerca el color al blanco o al negro hasta que se lea bien sobre el fondo
    local function readable(color, bg, need)
        local target = getContrastColor(bg)
        local c = color
        for i = 1, 10 do
            if contrast(c, bg) >= need then break end
            c = color:Lerp(target, i / 10)
        end
        return c
    end

    function Page.Guide(page, o)
        o = o or {}
        local pal = Kit.Palette
        local T = o.T or GetText
        local GUIDE, TAGS, TAG_COLORS = o.Data or {}, o.Tags or {}, o.TagColors or {}
        local painted = {}
        local function paintText(l)
            local role, bgRole = l:GetAttribute("YYGuideRole"), l:GetAttribute("YYGuideBg")
            local p = pal()
            local bg = p[bgRole] or p.Secondary
            local need = role == "TextDim" and 4.5 or 7
            l.TextColor3 = readable(p[role] or p.Text, bg, need)
            l.TextStrokeColor3 = bg
            l.TextStrokeTransparency = 0.75
        end
        local function txt(parent, es, en, size, font, role, props, bgRole)
            local l = new("TextLabel", {
                BackgroundTransparency = 1, Text = T(es, en), TextSize = size, Font = font,
                TextColor3 = pal()[role], TextXAlignment = Enum.TextXAlignment.Left, TextWrapped = true,
                Size = UDim2.new(1, 0, 0, 0), AutomaticSize = Enum.AutomaticSize.Y, ZIndex = 12,
            }, parent)
            l:SetAttribute("TextSpanish", es)
            l:SetAttribute("TextEnglish", en)
            l:SetAttribute("YYGuideRole", role)
            l:SetAttribute("YYGuideBg", bgRole or "Secondary")
            l:SetAttribute("IgnoreTextEffect", true)
            for k, v in pairs(props or {}) do l[k] = v end
            paintText(l)
            painted[#painted + 1] = l
            return l
        end
        local function repaintAll()
            local keep = {}
            for _, l in ipairs(painted) do
                if l.Parent then
                    pcall(paintText, l)
                    keep[#keep + 1] = l
                end
            end
            painted = keep
        end

        -- ---------- estructura ----------
        local p0 = pal()
        local root = new("Frame", {Name = "GuideRoot", Size = UDim2.new(1, 0, 0, 0), AutomaticSize = Enum.AutomaticSize.Y, BackgroundTransparency = 1, LayoutOrder = -100, ZIndex = 10}, page)

        new("UIPadding", {PaddingTop = UDim.new(0, 14), PaddingBottom = UDim.new(0, 14), PaddingLeft = UDim.new(0, 14), PaddingRight = UDim.new(0, 14)}, hero)
        new("UIListLayout", {Padding = UDim.new(0, 8), SortOrder = Enum.SortOrder.LayoutOrder}, hero)
        local heroTop = new("Frame", {Size = UDim2.new(1, 0, 0, 46), BackgroundTransparency = 1, LayoutOrder = 1, ZIndex = 11}, hero)
        local badge = new("Frame", {Size = UDim2.fromOffset(46, 46), BackgroundColor3 = p0.Accent, BorderSizePixel = 0, ZIndex = 11}, heroTop)
        badge:SetAttribute("ThemeRole", "Accent")
        round(badge, 14)
        local badgeGrad = new("UIGradient", {Rotation = 45, Transparency = NumberSequence.new(0, 0.35)}, badge)
        local badgeText = new("TextLabel", {Size = UDim2.fromScale(1, 1), BackgroundTransparency = 1, Text = o.Badge or "?", TextSize = 26, Font = Enum.Font.GothamBlack, TextColor3 = getContrastColor(p0.Accent), ZIndex = 12}, badge)
        badgeText:SetAttribute("IgnoreTextEffect", true)
        RealTweenService:Create(badgeGrad, TweenInfo.new(4, Enum.EasingStyle.Linear, Enum.EasingDirection.InOut, -1), {Rotation = 405}):Play()
        local title, sub = o.Title or {"GUÍA", "GUIDE"}, o.Subtitle or {"", ""}
        txt(heroTop, title[1], title[2], 17, Enum.Font.GothamBlack, "Text", {Position = UDim2.fromOffset(58, 3), Size = UDim2.new(1, -58, 0, 20), AutomaticSize = Enum.AutomaticSize.None, TextWrapped = false})
        txt(heroTop, sub[1], sub[2], 11, Enum.Font.GothamMedium, "TextDim", {Position = UDim2.fromOffset(58, 25), Size = UDim2.new(1, -58, 0, 16), AutomaticSize = Enum.AutomaticSize.None})

        -- buscador
        local ph = o.Placeholder or {"Buscar…", "Search…"}
        local searchBox = new("Frame", {Size = UDim2.new(1, 0, 0, 38), BackgroundColor3 = p0.Background, BackgroundTransparency = 0.25, BorderSizePixel = 0, LayoutOrder = 2, ZIndex = 11}, hero)
        searchBox:SetAttribute("ThemeRole", "Background")
        round(searchBox, 10)
        local searchStroke = Page.Stroke(searchBox, 0.55)
        local searchIcon = new("TextLabel", {Size = UDim2.fromOffset(34, 38), BackgroundTransparency = 1, Text = "🔍", TextSize = 15, Font = Enum.Font.GothamBold, TextColor3 = p0.TextDim, ZIndex = 12}, searchBox)
        searchIcon:SetAttribute("ThemeTextRole", "TextDim")
        searchIcon:SetAttribute("IgnoreTextEffect", true)
        local search = new("TextBox", {Size = UDim2.new(1, -44, 1, 0), Position = UDim2.fromOffset(34, 0), BackgroundTransparency = 1, Text = "", ClearTextOnFocus = false,
            PlaceholderText = T(ph[1], ph[2]), PlaceholderColor3 = p0.TextDim,
            TextColor3 = p0.Text, TextSize = 13, Font = Enum.Font.GothamBold, TextXAlignment = Enum.TextXAlignment.Left, ZIndex = 12}, searchBox)
        search:SetAttribute("ThemeTextRole", "Text")
        search:SetAttribute("PlaceholderSpanish", ph[1])
        search:SetAttribute("PlaceholderEnglish", ph[2])
        search.Focused:Connect(function() searchStroke.Color = pal().Accent searchStroke.Transparency = 0.05 end)
        search.FocusLost:Connect(function() searchStroke.Color = pal().Stroke searchStroke.Transparency = 0.55 end)

        -- chips de pestañas (desliza de costado)
        local chipScroll = new("ScrollingFrame", {Size = UDim2.new(1, 0, 0, 34), BackgroundTransparency = 1, BorderSizePixel = 0, ScrollBarThickness = 0, ScrollingDirection = Enum.ScrollingDirection.X,
            AutomaticCanvasSize = Enum.AutomaticSize.X, CanvasSize = UDim2.new(), LayoutOrder = 2, ZIndex = 10}, root)
        new("UIListLayout", {FillDirection = Enum.FillDirection.Horizontal, Padding = UDim.new(0, 6), SortOrder = Enum.SortOrder.LayoutOrder, VerticalAlignment = Enum.VerticalAlignment.Center}, chipScroll)

        local content = new("Frame", {Size = UDim2.new(1, 0, 0, 0), AutomaticSize = Enum.AutomaticSize.Y, BackgroundTransparency = 1, LayoutOrder = 3, ZIndex = 10}, root)
        new("UIListLayout", {Padding = UDim.new(0, 8), SortOrder = Enum.SortOrder.LayoutOrder}, content)

        -- ---------- piezas ----------
        local function tagPill(parent, key, order)
            local def = TAGS[key]
            if not def then return end
            local p = pal()
            local c = TAG_COLORS[key] or p.Accent
            local pill = new("TextLabel", {Size = UDim2.fromOffset(0, 18), AutomaticSize = Enum.AutomaticSize.X, BackgroundColor3 = c, BackgroundTransparency = 0.82,
                Text = T(def[1], def[2]), TextSize = 9, Font = Enum.Font.GothamBold, TextColor3 = Page.AdaptStatus(c), LayoutOrder = order, ZIndex = 13}, parent)
            pill:SetAttribute("TextSpanish", def[1])
            pill:SetAttribute("TextEnglish", def[2])
            pill:SetAttribute("IgnoreTextEffect", true)
            round(pill, 999)
            new("UIPadding", {PaddingLeft = UDim.new(0, 7), PaddingRight = UDim.new(0, 7)}, pill)
            new("UIStroke", {Color = c, Transparency = 0.5, Thickness = 1}, pill)
        end

        local function optionCard(parent, item, order, tabName)
            local p = pal()
            -- outer: fondo + toque; cardF (adentro): el contenido en lista (el botón no puede estar dentro de la lista)
            local outer = new("Frame", {Size = UDim2.new(1, 0, 0, 0), AutomaticSize = Enum.AutomaticSize.Y, BackgroundColor3 = p.Secondary, BackgroundTransparency = 0.35, BorderSizePixel = 0, LayoutOrder = order, ZIndex = 11}, parent)
            outer:SetAttribute("ThemeRole", "Secondary")
            round(outer, 12)
            local stroke = Page.Stroke(outer, 0.6)
            local cardF = new("Frame", {Size = UDim2.new(1, 0, 0, 0), AutomaticSize = Enum.AutomaticSize.Y, BackgroundTransparency = 1, ZIndex = 11}, outer)
            new("UIPadding", {PaddingTop = UDim.new(0, 10), PaddingBottom = UDim.new(0, 10), PaddingLeft = UDim.new(0, 12), PaddingRight = UDim.new(0, 12)}, cardF)
            new("UIListLayout", {Padding = UDim.new(0, 5), SortOrder = Enum.SortOrder.LayoutOrder}, cardF)

            local head = new("Frame", {Size = UDim2.new(1, 0, 0, 0), AutomaticSize = Enum.AutomaticSize.Y, BackgroundTransparency = 1, LayoutOrder = 1, ZIndex = 12}, cardF)
            txt(head, item.n[1], item.n[2], 14, Enum.Font.GothamBlack, "Text", {Size = UDim2.new(1, -30, 0, 0)})
            local chev = new("TextLabel", {AnchorPoint = Vector2.new(1, 0), Position = UDim2.new(1, 0, 0, -2), Size = UDim2.fromOffset(22, 22), BackgroundTransparency = 1, Text = "›", TextSize = 22, Font = Enum.Font.GothamBold, TextColor3 = p.TextDim, ZIndex = 12}, head)
            chev:SetAttribute("ThemeTextRole", "TextDim")
            chev:SetAttribute("IgnoreTextEffect", true)

            local tags = item.t or {}
            if tabName or #tags > 0 then
                local row = new("Frame", {Size = UDim2.new(1, 0, 0, 18), BackgroundTransparency = 1, LayoutOrder = 2, ZIndex = 12}, cardF)
                new("UIListLayout", {FillDirection = Enum.FillDirection.Horizontal, Padding = UDim.new(0, 4), SortOrder = Enum.SortOrder.LayoutOrder}, row)
                if tabName then
                    local tp = new("TextLabel", {Size = UDim2.fromOffset(0, 18), AutomaticSize = Enum.AutomaticSize.X, BackgroundColor3 = p.Text, BackgroundTransparency = 0.88, Text = tabName, TextSize = 9, Font = Enum.Font.GothamBlack, TextColor3 = p.Text, LayoutOrder = 0, ZIndex = 13}, row)
                    tp:SetAttribute("IgnoreTextEffect", true)
                    round(tp, 999)
                    new("UIPadding", {PaddingLeft = UDim.new(0, 7), PaddingRight = UDim.new(0, 7)}, tp)
                end
                for i, key in ipairs(tags) do tagPill(row, key, i) end
            end
            txt(cardF, item.s[1], item.s[2], 13, Enum.Font.GothamMedium, "TextDim", {LayoutOrder = 3})

            -- detalle (se abre al tocar)
            local detail = new("Frame", {Size = UDim2.new(1, 0, 0, 0), AutomaticSize = Enum.AutomaticSize.Y, BackgroundTransparency = 1, Visible = false, LayoutOrder = 4, ZIndex = 12}, cardF)
            new("UIListLayout", {Padding = UDim.new(0, 6), SortOrder = Enum.SortOrder.LayoutOrder}, detail)
            local hasDetail = false
            new("Frame", {Size = UDim2.new(1, 0, 0, 1), BackgroundColor3 = p.Stroke, BackgroundTransparency = 0.6, BorderSizePixel = 0, LayoutOrder = 0, ZIndex = 12}, detail)
            if item.h then
                hasDetail = true
                txt(detail, item.h[1], item.h[2], 13, Enum.Font.GothamMedium, "Text", {LayoutOrder = 1})
            end
            for i, s in ipairs(item.sl or {}) do
                hasDetail = true
                local line = new("Frame", {Size = UDim2.new(1, 0, 0, 0), AutomaticSize = Enum.AutomaticSize.Y, BackgroundTransparency = 1, LayoutOrder = 1 + i, ZIndex = 12}, detail)
                local key = new("TextLabel", {Size = UDim2.fromOffset(0, 20), AutomaticSize = Enum.AutomaticSize.X, BackgroundColor3 = p.Accent, BackgroundTransparency = 0, Text = s[1], TextSize = 11, Font = Enum.Font.GothamBold, TextColor3 = getContrastColor(p.Accent), ZIndex = 13}, line)
                key:SetAttribute("IgnoreTextEffect", true)
                round(key, 6)
                new("UIPadding", {PaddingLeft = UDim.new(0, 6), PaddingRight = UDim.new(0, 6)}, key)
                local desc = txt(line, s[2], s[3], 12, Enum.Font.GothamMedium, "TextDim", {Position = UDim2.fromOffset(0, 23)})
                desc.Size = UDim2.new(1, 0, 0, 0)
            end

            local open = false
            local hit = new("TextButton", {Size = UDim2.fromScale(1, 1), BackgroundTransparency = 1, Text = "", ZIndex = 20}, outer)
            hit.Activated:Connect(function()
                if not hasDetail then return end
                open = not open
                detail.Visible = open
                RealTweenService:Create(chev, TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {Rotation = open and 90 or 0}):Play()
                stroke.Color = open and pal().Accent or pal().Stroke
                stroke.Transparency = open and 0.2 or 0.6
                outer.BackgroundTransparency = open and 0.2 or 0.35
            end)
            if not hasDetail then chev.Visible = false end
            return outer
        end

        local function sectionTitle(parent, es, en, order)
            local f = new("Frame", {Size = UDim2.new(1, 0, 0, 22), BackgroundTransparency = 1, LayoutOrder = order, ZIndex = 11}, parent)
            local bar = new("Frame", {Size = UDim2.fromOffset(3, 14), Position = UDim2.fromOffset(2, 5), BackgroundColor3 = pal().Accent, BorderSizePixel = 0, ZIndex = 12}, f)
            bar:SetAttribute("ThemeRole", "Accent")
            round(bar, 999)
            txt(f, es, en, 12, Enum.Font.GothamBlack, "Text", {Position = UDim2.fromOffset(12, 3), Size = UDim2.new(1, -12, 0, 18), AutomaticSize = Enum.AutomaticSize.None, TextWrapped = false})
        end

        local function tabHeader(parent, g, order)
            local p = pal()
            local f = new("Frame", {Size = UDim2.new(1, 0, 0, 58), BackgroundColor3 = p.Secondary, BackgroundTransparency = 0.3, BorderSizePixel = 0, LayoutOrder = order, ZIndex = 11}, parent)
            f:SetAttribute("ThemeRole", "Secondary")
            round(f, 12)
            new("UIStroke", {Color = p.Accent, Transparency = 0.35, Thickness = 1.5}, f)
            local ic = new("ImageLabel", {Size = UDim2.fromOffset(30, 30), Position = UDim2.fromOffset(14, 14), BackgroundTransparency = 1, Image = g.Icon, ImageColor3 = p.Text, ScaleType = Enum.ScaleType.Fit, ZIndex = 12}, f)
            ic:SetAttribute("ThemeImageRole", "Text")
            txt(f, string.upper(g.Es), string.upper(g.En), 15, Enum.Font.GothamBlack, "Text", {Position = UDim2.fromOffset(54, 9), Size = UDim2.new(1, -64, 0, 18), AutomaticSize = Enum.AutomaticSize.None, TextWrapped = false})
            txt(f, g.Sum[1], g.Sum[2], 11, Enum.Font.GothamMedium, "TextDim", {Position = UDim2.fromOffset(54, 29), Size = UDim2.new(1, -64, 0, 22), AutomaticSize = Enum.AutomaticSize.None})
        end

        -- ---------- render ----------
        local Current = o.Start or (GUIDE[1] and GUIDE[1].Key)
        local chips = {}
        local function clear()
            for _, c in ipairs(content:GetChildren()) do
                if c:IsA("GuiObject") then c:Destroy() end
            end
        end
        local function renderTab(key)
            clear()
            local order = 0
            local function nextOrder() order = order + 1 return order end
            for _, g in ipairs(GUIDE) do
                if g.Key == key then
                    tabHeader(content, g, nextOrder())
                    for _, sec in ipairs(g.Sections) do
                        sectionTitle(content, sec[1], sec[2], nextOrder())
                        for _, item in ipairs(sec[3]) do
                            optionCard(content, item, nextOrder())
                        end
                    end
                end
            end
        end
        local function norm(s)
            s = string.lower(s or "")
            return (s:gsub("á", "a"):gsub("é", "e"):gsub("í", "i"):gsub("ó", "o"):gsub("ú", "u"):gsub("ñ", "n"))
        end
        local function renderSearch(q)
            clear()
            local nq = norm(q)
            local order, found = 0, 0
            for _, g in ipairs(GUIDE) do
                for _, sec in ipairs(g.Sections) do
                    for _, item in ipairs(sec[3]) do
                        local hay = norm(item.n[1] .. " " .. item.n[2] .. " " .. item.s[1] .. " " .. item.s[2])
                        if string.find(hay, nq, 1, true) then
                            order = order + 1
                            found = found + 1
                            optionCard(content, item, order, T(g.Es, g.En))
                        end
                    end
                end
            end
            if found == 0 then
                local es, en = "Sin resultados para \"" .. q .. "\".", "No results for \"" .. q .. "\"."
                if o.NotFound then es, en = o.NotFound(q) end
                txt(content, es, en, 12, Enum.Font.GothamMedium, "TextDim", {LayoutOrder = 1, TextXAlignment = Enum.TextXAlignment.Center})
            end
        end

        local function paintChips()
            local p = pal()
            for key, c in pairs(chips) do
                local on = key == Current and search.Text == ""
                c.BackgroundColor3 = on and p.Accent or p.Secondary
                c.BackgroundTransparency = on and 0 or 0.55
                c.TextColor3 = on and getContrastColor(p.Accent) or p.Text
            end
        end
        for i, g in ipairs(GUIDE) do
            local c = new("TextButton", {Size = UDim2.fromOffset(0, 28), AutomaticSize = Enum.AutomaticSize.X, BackgroundColor3 = p0.Secondary, Text = T(g.Es, g.En), TextSize = 12, Font = Enum.Font.GothamBold,
                TextColor3 = p0.Text, AutoButtonColor = false, LayoutOrder = i, ZIndex = 11}, chipScroll)
            c:SetAttribute("TextSpanish", g.Es)
            c:SetAttribute("TextEnglish", g.En)
            c:SetAttribute("IgnoreTextEffect", true)
            round(c, 999)
            new("UIPadding", {PaddingLeft = UDim.new(0, 14), PaddingRight = UDim.new(0, 14)}, c)
            chips[g.Key] = c
            c.Activated:Connect(function()
                Current = g.Key
                if search.Text ~= "" then search.Text = "" end
                renderTab(Current)
                paintChips()
                page.CanvasPosition = Vector2.new(0, 0)
            end)
        end

        local searchToken = 0
        search:GetPropertyChangedSignal("Text"):Connect(function()
            searchToken = searchToken + 1
            local token = searchToken
            task.delay(0.25, function()
                if token ~= searchToken then return end
                local q = search.Text:match("^%s*(.-)%s*$")
                if #q >= 2 then renderSearch(q) else renderTab(Current) end
                paintChips()
            end)
        end)

        renderTab(Current)
        paintChips()
        -- si cambia el tema, al volver a la guía se repintan chips y tarjetas
        page:GetPropertyChangedSignal("Visible"):Connect(function()
            if page.Visible then
                if search.Text == "" then renderTab(Current) end
                paintChips()
                repaintAll()
            end
        end)
        return {Root = root, Search = search}
    end
end
