-- ==================== SPRITE DE FONDO DE TEMA ====================
-- Un tema puede declarar Sprite = { SheetIds, Columns, Rows, FrameWidth,
-- FrameHeight, FramesPerSheet, FPS }. Tambien acepta el formato antiguo
-- { Ids, Columns, Rows, FrameWidth, FrameHeight, FrameCount, FPS }.
--
-- COMO FUNCIONA EL FONDO ANIMADO (SPRITE) DE UN TEMA
--
-- 1) DATOS (archivo de temas). Cada "hoja" (sheet) es una imagen subida a Roblox
--    como Image. Contiene una cuadricula de Columns x Rows frames de FrameWidth x
--    FrameHeight px. Orden de los frames: izquierda a derecha, arriba a abajo.
--    - SheetIds (o Ids): las hojas, en orden de reproduccion.
--    - FramesPerSheet: cuantos frames reales tiene cada hoja (la ultima suele quedar
--      incompleta). Con el formato antiguo basta FrameCount (total de frames).
--    - Una hoja de 4x5 celdas de 256x192 mide 1024x960 (Roblox limita a ~1024 px por lado).
--
-- 2) CARGA. LoadThemes guarda theme.Sprite en ThemeSprites[nombreDelTema] y lo borra
--    si el tema ya no lo trae.
--
-- 3) AL APLICAR UN TEMA, Window:SetTheme llama applyThemeSprite(self, BackgroundArt, sprite):
--    a) stopThemeSprite destruye las capas y la conexion del sprite anterior.
--    b) toThemeSpriteConfig valida el Sprite (acepta SheetIds o Ids) y calcula los
--       frames de cada hoja; si es invalido avisa con warn y se usa Background.
--    c) Se vacia BackgroundArt.Image y se crea UNA ImageLabel por hoja DENTRO de
--       BackgroundArt (createSpriteSheetLabels). Cada una ya tiene su Image y su
--       ImageRectSize (= tamano de UN frame). Solo la hoja actual se ve
--       (ImageTransparency 0); las demas estan en 1. Asi no se reasigna Image durante
--       el loop, lo que puede dar un parpadeo borroso al cambiar de hoja.
--    d) Las hojas se precargan con ContentProvider:PreloadAsync (en segundo plano).
--    e) Un RenderStepped acumula el tiempo y avanza 1 frame cada 1/FPS segundos
--       (el acumulado se limita a 0.25 s para no recuperar saltos despues de un lag).
--    f) updateSpriteSheetFrame localiza la hoja del frame, calcula
--         columna = indice % Columns,   fila = floor(indice / Columns)
--       y mueve ImageRectOffset = (columna * FrameWidth, fila * FrameHeight) en la hoja
--       activa (mas CropY si CropContent). Al terminar el ultimo frame vuelve al 0.
--
-- 4) FONDO ESTATICO. En SetTheme, applyImages no pone la imagen de Background mientras
--    haya un sprite activo (self._themeSprite). Si el tema no trae Sprite, o es
--    invalido, se usa Background como siempre.
--
-- 5) REGLA CRITICA - ZIndex. El ScreenGui usa ZIndexBehavior.Global: el orden de
--    dibujo lo decide el ZIndex en TODA la interfaz, no la jerarquia padre/hijo.
--    Las capas del sprite deben tener el MISMO ZIndex que BackgroundArt
--    (parent.ZIndex, hoy 5). Con ZIndex 1 el sprite quedaba debajo de Main (5) y
--    no se veia. Si algun dia cambia el ZIndex de BackgroundArt, el sprite lo sigue.
--
-- COMO AGREGAR UN TEMA CON SPRITE: en el archivo de temas, dentro del tema, agregar
--   Sprite = { SheetIds = {"rbxassetid://..."}, Columns = 4, Rows = 5,
--              FrameWidth = 256, FrameHeight = 192, FramesPerSheet = {20, 20, 8}, FPS = 24 }
-- y subir Version. Subir las hojas como Image y esperar la aprobacion de Roblox.

    pcall(function() btn.AutoButtonColor = false end)
    local c = btn:FindFirstChildOfClass("UICorner")
    if c then c.CornerRadius = UDim.new(0, 9) end
    if accent then
        btn.BackgroundColor3 = Theme.Accent
        btn:SetAttribute("ThemeRole", "Accent")
        if btn:IsA("TextButton") then
            btn.TextColor3 = Theme.AccentText or Color3.new(0, 0, 0)
            btn:SetAttribute("ThemeTextRole", "AccentText")
            btn.Font = Enum.Font.GothamBlack
            btn.TextSize = 15
        end
    else
        btn.BackgroundColor3 = Theme.Secondary
        btn.BackgroundTransparency = 0.2
        btn:SetAttribute("ThemeRole", "Secondary")
    end
    local scale = Instance.new("UIScale")
    scale.Parent = btn
    btn.MouseButton1Down:Connect(function()
        TweenService:Create(scale, TweenInfo.new(0.07), {Scale = 0.86}):Play()
    end)
    local function release()
        TweenService:Create(scale, TweenInfo.new(0.22, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {Scale = 1}):Play()
    end
    btn.MouseButton1Up:Connect(release)
    btn.MouseLeave:Connect(release)
end

function RowKit.Contrast(a, b)
    local function lum(c)
        local function ch(v) return v <= 0.03928 and v / 12.92 or ((v + 0.055) / 1.055) ^ 2.4 end
        return 0.2126 * ch(c.R) + 0.7152 * ch(c.G) + 0.0722 * ch(c.B)
    end
    local la, lb = lum(a), lum(b)
    if la < lb then la, lb = lb, la end
    return (la + 0.05) / (lb + 0.05)
end

-- Icono: si recibe un ID de Roblox usa la imagen (teñida con el color del tema); si no, el texto
function RowKit.Icon(parent, icon, px, z, role)
    role = role or "Accent"
    local id = tonumber(icon) or (type(icon) == "string" and tonumber(icon:match("^rbxassetid://(%d+)$")))
    if id then
        local img = mk("ImageLabel", {
            Parent = parent,
            Size = UDim2.fromOffset(px, px),
            AnchorPoint = Vector2.new(0.5, 0.5),
            Position = UDim2.new(0.5, 0, 0.5, 0),
            BackgroundTransparency = 1,
            Image = "rbxassetid://" .. tostring(id),
            ImageColor3 = Theme[role] or Color3.new(1, 1, 1),
            ScaleType = Enum.ScaleType.Fit,
            ZIndex = z,
        })
        img:SetAttribute("ThemeImageRole", role)
        return img
    end
    local t = mk("TextLabel", {
        Parent = parent,
        Size = UDim2.new(1, 0, 1, 0),
        BackgroundTransparency = 1,
        Text = tostring(icon or "◆"),
        TextSize = px,
        Font = Enum.Font.GothamBold,
        TextColor3 = Theme.Text,
        ZIndex = z,
    })
    t:SetAttribute("IgnoreTextEffect", true)
    return t
end

-- Selector segmentado (opciones excluyentes) con resaltado que se desliza
-- options = { {key=, es=, en=}, ... }  ->  devuelve { Set = function(key, animate) }
function RowKit.Segmented(page, options, currentKey, onSelect)
    local n = math.max(1, #options)
    local seg = {buttons = {}, current = nil}
    local holder = mk("Frame", {
        Parent = page,
        Size = UDim2.new(1, 0, 0, 42),
        BackgroundColor3 = Theme.Secondary,
        BackgroundTransparency = 0.4,
        BorderSizePixel = 0,
        ZIndex = 9,
    })
    holder:SetAttribute("ThemeRole", "Secondary")
    corner(holder, 14)
    stroke(holder, Theme.Stroke, 1, 0.6)
    local pill = mk("Frame", {
        Parent = holder,
        Size = UDim2.new(1 / n, -6, 1, -6),
        Position = UDim2.new(0, 3, 0, 3),
        BackgroundColor3 = Theme.Accent,
        BorderSizePixel = 0,
        ZIndex = 10,
    })
    pill:SetAttribute("ThemeRole", "Accent")
    corner(pill, 11)
    mk("UIGradient", {Parent = pill, Rotation = 90, Color = ColorSequence.new(Color3.new(1, 1, 1), Color3.fromRGB(215, 215, 215))})
    seg.Holder = holder

    for i, opt in ipairs(options) do
        local b = mk("TextButton", {
            Parent = holder,
            Size = UDim2.new(1 / n, 0, 1, 0),
            Position = UDim2.new((i - 1) / n, 0, 0, 0),
            BackgroundTransparency = 1,
            Text = GetText(opt.es, opt.en),
            TextColor3 = Theme.TextDim,
            Font = Enum.Font.GothamBold,
            TextSize = 12,
            TextTruncate = Enum.TextTruncate.AtEnd,
            AutoButtonColor = false,
            ZIndex = 11,
        })
        b:SetAttribute("TextSpanish", opt.es)
        b:SetAttribute("TextEnglish", opt.en)
        b:SetAttribute("IgnoreTextEffect", true)
        seg.buttons[i] = {key = opt.key, button = b}
        b.MouseButton1Click:Connect(function()
            if seg.current == opt.key then return end
            playSound(Sounds.Click, 0.45)
            seg.Set(opt.key, true)
            if onSelect then onSelect(opt.key) end
        end)
    end

    function seg.Set(key, animate)
        for i, entry in ipairs(seg.buttons) do
            local on = entry.key == key
            if on then
                seg.current = key
                local goal = {Position = UDim2.new((i - 1) / n, 3, 0, 3)}
                if animate then
                    TweenService:Create(pill, TweenInfo.new(0.28, Enum.EasingStyle.Quint, Enum.EasingDirection.Out), goal):Play()
                else
                    pill.Position = goal.Position
                end
            end
            entry.button.TextColor3 = on and Theme.AccentText or Theme.TextDim
            entry.button:SetAttribute("ThemeTextRole", on and "AccentText" or "TextDim")
            entry.button.Font = on and Enum.Font.GothamBlack or Enum.Font.GothamBold
        end
    end
    seg.Set(currentKey, false)
    return seg
end
